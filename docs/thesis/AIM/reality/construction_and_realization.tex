\chapter{Construction and Physical Realization}

\label{chap:construction-and-realization}

This chapter answers how target engineering state becomes physically
built railway state under construction and maintenance constraints. It
establishes the realization operator view used in Reality and clarifies
the consequence for runtime reconstruction and realization-aware design.

The opening question is therefore comparative: what changes when
analytic target state is transformed into realized state?

\end{principle}

\section{Physical Realization versus Metric Realization}
% ============================================================

Metric realization (\cref{chap:metric-realization}) defines measurable
embedding context. Physical realization describes construction and
maintenance effects on built infrastructure. The two layers are coupled
but distinct.

\begin{definition}[Realized alignment]
\label{def:realized-alignment}
The realized alignment is the physically constructed railway geometry,
denoted by
\[
\gamma_{\mathrm{real}}(s).
\]
\end{definition}

\begin{definition}[Realized curvature]
\label{def:realized-curvature}
The realized curvature is
\[
\kappa_{\mathrm{real}}(s)=\frac{\dd\theta_{\mathrm{real}}}{\dd s}.
\]
\end{definition}

\section{Construction and Adjustment Process}
% ============================================================

Construction and maintenance are iterative local processes.

The following definitions and propositions formalize this practical
workflow as an iterative realization process whose fixed-point view
connects engineering practice to operator semantics.

\begin{definition}[Adjustment operator]
\label{def:adjustment-operator}
The adjustment operator \(\mathcal A\) denotes a localized
construction/maintenance update applied to an existing alignment state.
\end{definition}

\begin{definition}[Iterative realization]
\label{def:iterative-construction-realization}
Starting from \(\gamma_0\), an iterative realization process can be
represented by
\[
\gamma_{k+1}=\mathcal A_{\gamma_{\mathrm{target}}}(\gamma_k).
\]
\end{definition}

\begin{definition}[Realized alignment as limit]
\label{def:realized-alignment-limit}
If this process converges,
\[
\gamma_{\mathrm{real}}=\lim_{k\to\infty}\gamma_k.
\]
\end{definition}

\begin{proposition}[Fixed point interpretation]
\label{prop:realization-fixed-point}
If iterative realization converges,
\[
\gamma_{\mathrm{real}}=
\mathcal A_{\gamma_{\mathrm{target}}}(\gamma_{\mathrm{real}}).
\]
\end{proposition}

\section{Sampling, Locality, and Filtering}
% ============================================================

\begin{definition}[Control points]
\label{def:control-points}
Let
\[
0=s_0<s_1<\dots<s_N=L
\]
be control-point stations used in realization and maintenance workflows.
\end{definition}

\begin{proposition}[Adaptive sampling principle]
\label{prop:adaptive-sampling-principle}
Sampling density should increase in regions with strong curvature
variation.
\end{proposition}

\begin{definition}[Local correction]
\label{def:local-correction}
A local correction form is
\[
\mathcal A_{\gamma_{\mathrm{target}}}(\gamma)
=
\gamma+\Delta(\gamma,\gamma_{\mathrm{target}}).
\]
\end{definition}

\begin{proposition}[Locality]
\label{prop:adjustment-locality}
Adjustment depends on local neighbourhood context rather than global
alignment state.
\end{proposition}

\begin{definition}[Kernel-based adjustment]
\label{def:kernel-based-adjustment}
A linearized local model is
\[
(\mathcal A_{\gamma_{\mathrm{target}}}\gamma)(s)
=
\gamma(s)+
\int_I K(s,\sigma)
\bigl(\gamma_{\mathrm{target}}(\sigma)-\gamma(\sigma)\bigr)
\,\dd\sigma.
\]
\end{definition}

\begin{definition}[Discrete adjustment model]
\label{def:discrete-adjustment-model}
For control points,
\[
\gamma_i^{(k+1)}
=
\gamma_i^{(k)}+
\sum_j K_{ij}\left(\gamma_j^{\mathrm{target}}-\gamma_j^{(k)}\right).
\]
\end{definition}

\begin{proposition}
\label{prop:realization-low-pass}
Physical realization acts approximately as a low-pass filter on
curvature and related high-frequency components.
\end{proposition}

\begin{principle}[Realization filter principle]
\label{princ:realization-filter}
Construction and maintenance transform target state through physical
sampling, machine response, and structural filtering.
\end{principle}

\section{Runtime and Optimization Implications}
% ============================================================

The realization layer is compatible with sparse and runtime evaluation:
realized geometry can be interpreted as measured samples, reconstructed
runtime state, or sparse-compatible approximation.

This section therefore transitions from construction mechanics to
engineering consequence: realized-state performance becomes the relevant
design criterion for downstream optimization.

\begin{principle}[Sparse realization compatibility]
\label{princ:sparse-realization-compatibility}
Realization-aware workflows must preserve compatibility with sparse and
runtime state architectures.
\end{principle}

\begin{definition}[Realization-aware objective]
\label{def:realization-aware-objective}
A realization-aware design objective can be written as
\[
\min_{x_{\mathrm{target}}}J\bigl(\mathcal R(x_{\mathrm{target}})\bigr).
\]
\end{definition}

\begin{proposition}[Realization-aware design principle]
\label{prop:realization-aware-design}
The relevant design criterion is performance of realized state, not only
analytic smoothness of target state.
\end{proposition}

\begin{proposition}
\label{prop:physical-design-space}
The physical design space is the image of the realization operator.
\end{proposition}

Construction and physical realization connect target engineering state
