\chapter{Realized Pose3 and Dynamic Realization}

\label{chap:realized-pose3}

% ============================================================
\section*{Motivation}
% ============================================================

\begin{remark}
Classical alignment theory typically assumes that the designed geometry
is identical to the realized geometry.
\end{remark}

\begin{remark}
In reality, railway tracks are produced through iterative and physically
constrained realization processes.
\end{remark}

\begin{remark}
As a consequence, not only the geometric position of the track differs
from the theoretical target alignment, but also its orientation,
curvature, cant, and dynamic behaviour.
\end{remark}

\begin{remark}
Thus, realization affects the full spatial railway state rather than
geometry alone.
\end{remark}

\begin{remark}
This motivates the introduction of a realized pose3 state:
\[
\mathrm{pose3}_{\mathrm{real}}(s).
\]
\end{remark}

\begin{principle}[Realized-state principle]
\label{princ:realized-state-principle}
Within AIM, the physically relevant railway state is the realized runtime
state, not merely the analytical target state.
\end{principle}

% ============================================================
\section{Target and Realized States}
% ============================================================

\begin{definition}[Target pose3]
\label{def:target-pose3}
The target railway state is defined as
\[
\mathrm{pose3}_{\mathrm{target}}(s)
=
\bigl(
\Gamma_{\mathrm{target}}(s),
F_{\mathrm{rail,target}}(s),
\phi_{\mathrm{target}}(s)
\bigr).
\]
\end{definition}

\begin{definition}[Realized pose3]
\label{def:realized-pose3}
The realized railway state is defined as
\[
\mathrm{pose3}_{\mathrm{real}}(s)
=
\bigl(
\Gamma_{\mathrm{real}}(s),
F_{\mathrm{rail,real}}(s),
\phi_{\mathrm{real}}(s)
\bigr).
\]
\end{definition}

\begin{remark}
The realized state is the physically constructed, measurable, and
operationally effective state of the railway track.
\end{remark}

\begin{remark}
In practice,
\[
\mathrm{pose3}_{\mathrm{real}}
\neq
\mathrm{pose3}_{\mathrm{target}}.
\]
\end{remark}

\begin{remark}
The difference is not limited to position. It affects tangent direction,
curvature, cant, frame orientation, and derived dynamic quantities.
\end{remark}

% ============================================================
\section{Measured States}
% ============================================================

\begin{definition}[Measured pose3 state]
\label{def:measured-pose3}
A measured pose3 state is an observation-derived approximation of the
realized railway state:
\[
\mathrm{pose3}_{\mathrm{meas}}(s_i)
\approx
\mathrm{pose3}_{\mathrm{real}}(s_i).
\]
\end{definition}

\begin{remark}
Measured pose3 states are usually available only at discrete stations,
timestamps, or sensor positions.
\end{remark}

\begin{remark}
Measurement may provide direct or indirect information about:
\begin{itemize}
\item position,
\item tangent direction,
\item curvature,
\item cant,
\item gauge,
\item vertical profile,
\item acceleration,
\item and local frame orientation.
\end{itemize}
\end{remark}

\begin{remark}
Measured states are not identical to realized states. They include sensor
noise, sampling effects, calibration errors, filtering effects, and
interpretation assumptions.
\end{remark}

\begin{principle}[Measurement separation principle]
\label{princ:measurement-separation}
AIM distinguishes target state, realized state, and measured state as
separate conceptual layers.
\end{principle}

% ============================================================
\section{Sources of Realization Deviations}
% ============================================================

\begin{remark}
Deviations between target and realized state arise from:
\begin{itemize}
\item measurement uncertainty,
\item machine limitations,
\item discrete control points,
\item structural elasticity,
\item ballast behaviour,
\item rail bending,
\item settlement processes,
\item maintenance history,
\item and operational degradation.
\end{itemize}
\end{remark}

\begin{remark}
Thus, realization is not a purely geometric transformation, but a
mechanical and operational process.
\end{remark}

% ============================================================
\section{Realized Geometry}
% ============================================================

\begin{definition}[Realized spatial trajectory]
\label{def:realized-spatial-trajectory}
The realized spatial trajectory is denoted by:
\[
\Gamma_{\mathrm{real}}(s).
\]
\end{definition}

\begin{remark}
\(\Gamma_{\mathrm{real}}(s)\) represents the physically realized railway
reference trajectory after construction, maintenance, settlement, and
operational loading.
\end{remark}

\begin{remark}
It may be represented as:
\begin{itemize}
\item discrete measured samples,
\item a reconstructed dense curve,
\item a sparse re-fit,
\item or a hybrid measured--analytical object.
\end{itemize}
\end{remark}

\begin{remark}
Within AIM, realized geometry remains compatible with sparse modelling
by being interpreted either as reconstructed runtime state or as a
sparse-compatible approximation.
\end{remark}

% ============================================================
\section{Runtime Reconstruction}
% ============================================================

\begin{remark}
Realized pose3 is generally not stored directly as a complete continuous
object.
\end{remark}

\begin{remark}
Instead, it is reconstructed at runtime from:
\begin{itemize}
\item measured samples,
\item sparse target geometry,
\item realization models,
\item frame operators,
\item cant reconstruction,
\item profile reconstruction,
\item and metric context.
\end{itemize}
\end{remark}

\begin{definition}[Runtime realization reconstruction]
\label{def:runtime-realization-reconstruction}
Runtime reconstruction of realized pose3 may be written abstractly as:
\[
\mathrm{pose3}_{\mathrm{real}}(s)
=
\mathcal E_{\mathrm{real}}
\bigl(
s,
x_{\mathrm{target}},
m,
\mathcal R,
\mathcal C
\bigr),
\]
where \(m\) denotes measured data, \(\mathcal R\) denotes realization
model information, and \(\mathcal C\) denotes coordinate and metric
context.
\end{definition}

\begin{remark}
The operator \(\mathcal E_{\mathrm{real}}\) is a runtime evaluation
operator rather than a persistent storage object.
\end{remark}

% ============================================================
\section{Realized Curvature and Cant}
% ============================================================

\begin{definition}[Realized curvature]
\label{def:realized-pose3-curvature}
The realized curvature is:
\[
\kappa_{\mathrm{real}}(s)
=
\frac{\dd \theta_{\mathrm{real}}}{\dd s}.
\]
\end{definition}

\begin{definition}[Realized cant]
\label{def:realized-cant}
The realized cant angle is:
\[
\phi_{\mathrm{real}}(s).
\]
\end{definition}

\begin{remark}
Both quantities differ from their target values due to realization
effects.
\end{remark}

\begin{remark}
In particular,
\[
\kappa_{\mathrm{real}}
\neq
\kappa_{\mathrm{target}},
\qquad
\phi_{\mathrm{real}}
\neq
\phi_{\mathrm{target}}.
\]
\end{remark}

\begin{remark}
Measured curvature and measured cant are further approximations:
\[
\kappa_{\mathrm{meas}}
\approx
\kappa_{\mathrm{real}},
\qquad
\phi_{\mathrm{meas}}
\approx
\phi_{\mathrm{real}}.
\]
\end{remark}

% ============================================================
\section{Cant Realization}
% ============================================================

\begin{remark}
Cant realization is particularly sensitive because cant is not only a
geometric quantity but also a dynamically relevant operational state.
\end{remark}

\begin{remark}
Cant realization is affected by:
\begin{itemize}
\item tamping accuracy,
\item rail fastening systems,
\item ballast stiffness,
\item torsion limits,
\item local support conditions,
\item measurement procedures,
\item and sensor interpretation.
\end{itemize}
\end{remark}

\begin{remark}
In practice, cant ramps are realized only approximately and may exhibit
local irregularities.
\end{remark}

\begin{remark}
This is especially relevant in:
\begin{itemize}
\item transition zones,
\item switches and crossings,
\item station areas,
\item overlapping transition regions,
\item and locally constrained construction zones.
\end{itemize}
\end{remark}

% ============================================================
\section{Measurement Ambiguity}
% ============================================================

\begin{remark}
Measured railway data does not automatically define a unique realized
pose3 state.
\end{remark}

\begin{remark}
Ambiguities may arise from:
\begin{itemize}
\item uncertain station assignment,
\item CRS uncertainty,
\item projection ambiguity,
\item sensor offset from the reference line,
\item vehicle motion,
\item filtering by onboard systems,
\item incomplete cant or profile information,
\item and topology ambiguity in multi-track environments.
\end{itemize}
\end{remark}

\begin{remark}
Therefore, measurement interpretation is itself a runtime reconstruction
problem.
\end{remark}

\begin{principle}[Measured-state ambiguity]
\label{princ:measured-state-ambiguity}
Measured data must be interpreted as evidence for realized pose3, not as
realized pose3 itself.
\end{principle}

% ============================================================
\section{Sensor Integration}
% ============================================================

\begin{remark}
Different sensors observe different projections of the realized railway
state.
\end{remark}

\begin{remark}
Examples include:
\begin{itemize}
\item surveying points,
\item track geometry cars,
\item inertial measurement units,
\item GNSS receivers,
\item laser scanning,
\item photogrammetry,
\item gauge and cant measurement systems,
\item and vehicle-mounted acceleration sensors.
\end{itemize}
\end{remark}

\begin{remark}
Sensor integration requires mapping heterogeneous observations into a
common runtime state interpretation.
\end{remark}

\begin{definition}[Sensor observation model]
\label{def:sensor-observation-model}
A sensor observation may be written as:
\[
y_i
=
\mathcal H_i
\bigl(
\mathrm{pose3}_{\mathrm{real}}(s_i)
\bigr)
+
\varepsilon_i,
\]
where \(\mathcal H_i\) denotes the sensor observation operator and
\(\varepsilon_i\) denotes measurement error.
\end{definition}

\begin{remark}
The reconstruction task is therefore an inverse problem: estimate the
realized pose3 state from heterogeneous noisy observations.
\end{remark}

% ============================================================
\section{Coupling of Curvature and Cant}
% ============================================================

\begin{remark}
Curvature and cant are dynamically coupled quantities.
\end{remark}

\begin{remark}
Therefore, realization errors in curvature and cant cannot be treated
independently.
\end{remark}

\begin{remark}
The physically relevant quantity is the realized dynamic state:
\[
(\kappa_{\mathrm{real}}, \phi_{\mathrm{real}}).
\]
\end{remark}

\begin{definition}[Realized equilibrium condition]
\label{def:realized-equilibrium-condition}
A realized track is dynamically balanced if:
\[
v^2\kappa_{\mathrm{real}}(s)
=
g\sin\phi_{\mathrm{real}}(s).
\]
\end{definition}

\begin{remark}
In practice, deviations from equilibrium are unavoidable and constrained
only within admissible limits.
\end{remark}

% ============================================================
\section{Dynamic Realization Error}
% ============================================================

\begin{definition}[Realized effective lateral acceleration]
\label{def:realized-effective-lateral-acceleration}
The realized effective lateral acceleration is:
\[
a_{\mathrm{eff,real}}(s)
=
v^2\kappa_{\mathrm{real}}(s)
-
g\sin\phi_{\mathrm{real}}(s).
\]
\end{definition}

\begin{remark}
This quantity is operationally more relevant than geometric deviation
alone.
\end{remark}

\begin{definition}[Realized jerk]
\label{def:realized-jerk}
The realized jerk is:
\[
j_{\mathrm{real}}(s)
=
v^3\kappa'_{\mathrm{real}}(s)
-
gv\cos\phi_{\mathrm{real}}(s)\,
\phi'_{\mathrm{real}}(s).
\]
\end{definition}

\begin{remark}
Passenger comfort depends directly on the realized jerk rather than on
the analytical target functions.
\end{remark}

% ============================================================
\section{Realization as State Transformation}
% ============================================================

\begin{definition}[Pose3 realization operator]
\label{def:pose3-realization-operator}
The realization process may be written as:
\[
\mathcal{R}_{\mathrm{pose3}}
:
\mathrm{pose3}_{\mathrm{target}}
\mapsto
\mathrm{pose3}_{\mathrm{real}}.
\]
\end{definition}

\begin{remark}
This operator includes:
\begin{itemize}
\item geometric realization,
\item cant realization,
\item profile realization,
\item mechanical filtering,
\item measurement feedback,
\item and operational constraints.
\end{itemize}
\end{remark}

\begin{remark}
Thus, realization acts on the full railway state rather than only on
position space.
\end{remark}

\begin{remark}
In practice, \(\mathcal{R}_{\mathrm{pose3}}\) is not directly known. It is
approximated through construction knowledge, measurement data, machine
models, and runtime reconstruction.
\end{remark}

% ============================================================
\section{Filtering Interpretation}
% ============================================================

\begin{proposition}
The realization operator acts as a low-pass filter on:
\begin{itemize}
\item curvature,
\item cant,
\item profile irregularities,
\item and higher-order derivatives.
\end{itemize}
\end{proposition}

\begin{remark}
High-frequency components are attenuated through:
\begin{itemize}
\item machine limitations,
\item structural smoothing,
\item ballast mechanics,
\item vehicle-track interaction,
\item and measurement discretization.
\end{itemize}
\end{remark}

\begin{remark}
Therefore, highly detailed analytical target states may collapse into
similar realized states.
\end{remark}

% ============================================================
\section{Operational Interpretation}
% ============================================================

\begin{remark}
Railway operation interacts with the realized state rather than with the
theoretical target geometry.
\end{remark}

\begin{remark}
Consequently, the relevant engineering question is not:
\begin{quote}
``Was the theoretical geometry built exactly?''
\end{quote}
but rather:
\begin{quote}
``Does the realized state produce the intended dynamic behaviour?''
\end{quote}
\end{remark}

\begin{remark}
This makes realized pose3 the natural bridge between construction,
measurement, runtime evaluation, and operation.
\end{remark}

% ============================================================
\section{Connection to Optimization}
% ============================================================

\begin{remark}
Optimization should therefore operate on realized states:
\[
J
=
J\bigl(
\mathrm{pose3}_{\mathrm{real}}
\bigr)
=
J\bigl(
\mathcal{R}_{\mathrm{pose3}}
(
\mathrm{pose3}_{\mathrm{target}}
)
\bigr).
\]
\end{remark}

\begin{remark}
This extends classical alignment optimization toward realization-aware
optimization.
\end{remark}

\begin{remark}
When measured states are available, optimization may additionally use:
\[
J
=
J\bigl(
\mathrm{pose3}_{\mathrm{real}},
\mathrm{pose3}_{\mathrm{meas}}
\bigr),
\]
to calibrate realization models or identify maintenance needs.
\end{remark}

% ============================================================
\section{Connection to AIM}
% ============================================================

\begin{summary}
Real railway tracks are not analytical curves but realized physical
states.

The realization process acts on the complete pose3 state, including
geometry, curvature, profile, frame orientation, and cant.

Measured railway data provides evidence for this realized state, but it
does not define it uniquely. Therefore, AIM distinguishes target pose3,
realized pose3, and measured pose3 as separate layers.

Within the AIM framework, realized pose3 forms the bridge between:
\begin{itemize}
\item analytical geometry,
\item runtime reconstruction,
\item sensor integration,
\item construction and maintenance,
\item physical realization,
\item and operational dynamics.
\end{itemize}

Thus, railway alignment design must be understood as the design of a
realizable dynamic state rather than merely a geometric target curve.
\end{summary}
