\chapter{Realized pose3 and Target--Realized Comparison}

\label{chap:realized-pose3}

This chapter consumes the canonical pose3 definitions from Foundations
and State (\cref{def:pose3,def:runtime-pose3}) and defines the realized
and measured counterparts used in Reality workflows.

\begin{principle}[Realized-state principle]
\label{princ:realized-state-principle}
Within AIM, operational engineering interpretation depends on realized
runtime state rather than on target state alone.
\end{principle}

% ============================================================
\section{Target, Realized, and Measured pose3}
% ============================================================

\begin{definition}[Target pose3]
\label{def:target-pose3}
The target railway state is denoted by
\[
\mathrm{pose3}_{\mathrm{target}}(s).
\]
\end{definition}

\begin{definition}[Realized pose3]
\label{def:realized-pose3}
The realized railway state is denoted by
\[
\mathrm{pose3}_{\mathrm{real}}(s).
\]
\end{definition}

\begin{definition}[Measured pose3 state]
\label{def:measured-pose3}
A measured pose3 state is an observation-derived approximation
\[
\mathrm{pose3}_{\mathrm{meas}}(s_i)
\approx
\mathrm{pose3}_{\mathrm{real}}(s_i).
\]
\end{definition}

\begin{principle}[Measurement separation principle]
\label{princ:measurement-separation}
Target state, realized state, and measured state are distinct layers.
\end{principle}

% ============================================================
\section{Realized State Components}
% ============================================================

\begin{definition}[Realized spatial trajectory]
\label{def:realized-spatial-trajectory}
The realized spatial trajectory is denoted by
\[
\Gamma_{\mathrm{real}}(s).
\]
\end{definition}

\begin{definition}[Realized curvature]
\label{def:realized-pose3-curvature}
\[
\kappa_{\mathrm{real}}(s)=\frac{\dd\theta_{\mathrm{real}}}{\dd s}.
\]
\end{definition}

\begin{definition}[Realized cant]
\label{def:realized-cant}
\[
\phi_{\mathrm{real}}(s).
\]
\end{definition}

In general,
\[
\mathrm{pose3}_{\mathrm{real}}
\neq
\mathrm{pose3}_{\mathrm{target}}.
\]

% ============================================================
\section{Runtime Reconstruction of Realized pose3}
% ============================================================

Realized pose3 is typically reconstructed at runtime from observed data,
realization models, and embedding context.

\begin{definition}[Runtime realization reconstruction]
\label{def:runtime-realization-reconstruction}
A schematic form is
\[
\mathrm{pose3}_{\mathrm{real}}(s)
=
\mathcal E_{\mathrm{real}}(s,x_{\mathrm{target}},m,\mathcal R,\mathcal C),
\]
where \(m\) is observation data, \(\mathcal R\) realization-model
information, and \(\mathcal C\) metric/coordinate context.
\end{definition}

\begin{principle}[Measured-state ambiguity]
\label{princ:measured-state-ambiguity}
Measured data is evidence for realized pose3, not realized pose3 itself.
\end{principle}

% ============================================================
\section{Observation Operators and Sensor Integration}
% ============================================================

\begin{definition}[Sensor observation model]
\label{def:sensor-observation-model}
\[
y_i
=
\mathcal H_i\bigl(\mathrm{pose3}_{\mathrm{real}}(s_i)\bigr)
+
\varepsilon_i,
\]
where \(\mathcal H_i\) is a sensor observation operator and
\(\varepsilon_i\) observation error.
\end{definition}

Realized-state inference is therefore an inverse reconstruction problem
from heterogeneous observations.

% ============================================================
\section{Dynamic Comparison Quantities}
% ============================================================

\begin{definition}[Realized equilibrium condition]
\label{def:realized-equilibrium-condition}
A realized track is dynamically balanced if
\[
v^2\kappa_{\mathrm{real}}(s)=g\sin\phi_{\mathrm{real}}(s).
\]
\end{definition}

\begin{definition}[Realized effective lateral acceleration]
\label{def:realized-effective-lateral-acceleration}
\[
a_{\mathrm{eff,real}}(s)
=
v^2\kappa_{\mathrm{real}}(s)-g\sin\phi_{\mathrm{real}}(s).
\]
\end{definition}

\begin{definition}[Realized jerk]
\label{def:realized-jerk}
\[
j_{\mathrm{real}}(s)
=
v^3\kappa'_{\mathrm{real}}(s)-gv\cos\phi_{\mathrm{real}}(s)\,\phi'_{\mathrm{real}}(s).
\]
\end{definition}

These quantities support target--realized engineering comparison beyond
position-only deviation.

% ============================================================
\section{Realization Operator on pose3}
% ============================================================

\begin{definition}[Pose3 realization operator]
\label{def:pose3-realization-operator}
\[
\mathcal R_{\mathrm{pose3}}:
\mathrm{pose3}_{\mathrm{target}}
\mapsto
\mathrm{pose3}_{\mathrm{real}}.
\]
\end{definition}

This operator acts on full railway state interpretation (geometry,
frame, cant, and derived dynamic behaviour), not on coordinates alone.

% ============================================================
\section{Connection to AIM}
% ============================================================

\begin{summary}
Realized pose3 is the central comparison layer between target
engineering state and observed railway reality.

Within AIM, the chapter preserves separation between target, realized,
and measured state while consuming canonical pose3 and realization
operators from prior parts.
\end{summary}
