\chapter{Notation and Terminology}

\label{chap:notation}

% ============================================================
\section{Motivation}
% ============================================================

\begin{remark}
Railway alignment engineering combines concepts from geometry,
surveying, dynamics, construction, realization, runtime evaluation, and
operational railway systems.
\end{remark}

\begin{remark}
Many inconsistencies in classical railway systems arise not from
incorrect mathematics, but from ambiguous terminology and mixed
conceptual layers.
\end{remark}

\begin{remark}
The AIM framework therefore introduces a canonical notation and
terminology system.
\end{remark}

\begin{remark}
The purpose of this chapter is:
\begin{itemize}
\item to define canonical symbols,
\item to stabilize terminology,
\item to reduce ambiguity,
\item and to separate fundamentally different conceptual layers.
\end{itemize}
\end{remark}

% ============================================================
\section{General Principles}
% ============================================================

\begin{principle}[Intrinsic-first principle]
\label{princ:intrinsic-first}

Whenever possible, railway geometry is formulated intrinsically using
arc-length-based quantities.

\end{principle}

% ------------------------------------------------------------

\begin{principle}[Runtime distinction principle]
\label{princ:runtime-distinction}

A distinction is made between:
\begin{itemize}
\item stored representations,
\item runtime-evaluated states,
\item and physically realized states.
\end{itemize}

\end{principle}

% ------------------------------------------------------------

\begin{principle}[Target vs realization principle]
\label{princ:target-vs-realization}

Target geometry and realized geometry are treated as distinct objects.

\end{principle}

% ------------------------------------------------------------

\begin{principle}[Canonical layer principle]
\label{princ:canonical-layer}

AIM separates:
\begin{itemize}
\item intrinsic geometry,
\item metric realization,
\item runtime interpretation,
\item physical realization,
\item operational behaviour,
\item and CRS-dependent embedding
\end{itemize}
into distinct conceptual layers.

\end{principle}

\section{Canonical Coordinates}
% ============================================================

\subsection{Arc-Length}

\begin{definition}[Arc-length]
\label{def:arc-length}

\[
s
\]

denotes intrinsic arc-length along the alignment.

\end{definition}

% ------------------------------------------------------------

\begin{remark}
Arc-length is the primary intrinsic coordinate within AIM.
\end{remark}

% ============================================================

\subsection{Track Coordinates}

\begin{definition}[Track coordinates]
\label{def:track-coordinates}

Track-space coordinates are:
\[
(s,d,h),
\]
where:
\begin{itemize}
\item \(s\): longitudinal coordinate,
\item \(d\): lateral offset,
\item \(h\): vertical offset.
\end{itemize}

\end{definition}

% ------------------------------------------------------------

\begin{remark}
Track coordinates belong to \(\Omega_{\mathrm{track}}\).
\end{remark}

% ------------------------------------------------------------

\begin{remark}
Additional operational coordinates may be introduced by runtime systems.
\end{remark}

% ============================================================

\subsection{World Coordinates}

\begin{definition}[World coordinates]
\label{def:world-coordinates}

World coordinates are denoted by:
\[
x \in \Omega_{\mathrm{world}}.
\]

\end{definition}

% ------------------------------------------------------------

\begin{remark}
World coordinates are CRS-dependent.
\end{remark}

% ------------------------------------------------------------

\begin{remark}
When required, world-space quantities may explicitly use the notation:
\[
x_{\mathrm w}.
\]
\end{remark}

\section{Geometry Symbols}
% ============================================================

\subsection{Alignment Curve}

\begin{definition}[Alignment curve]
\label{def:alignment-curve}

\[
\gamma(s)
\]

denotes the intrinsic alignment geometry.

\end{definition}

% ------------------------------------------------------------

\begin{remark}
The alignment curve represents the geometric backbone of the railway
alignment.
\end{remark}

% ------------------------------------------------------------

\begin{remark}
Depending on context, \(\gamma(s)\) may denote planar geometry, spatial
embedding, target geometry, or realized geometry. Such distinctions
should be marked explicitly when relevant.
\end{remark}

% ============================================================

\subsection{Orientation}

\begin{definition}[Orientation angle]
\label{def:orientation-angle}

\[
\theta(s)
\]

denotes horizontal alignment orientation.

\end{definition}

% ============================================================

\subsection{Curvature}

\begin{definition}[Curvature]
\label{def:curvature}

\[
\kappa(s)
\]

denotes curvature as function of arc-length.

\end{definition}

% ------------------------------------------------------------

\begin{remark}
Curvature is the primary intrinsic geometric quantity within AIM.
\end{remark}

% ============================================================

\subsection{Cant}

\begin{definition}[Cant angle]
\label{def:cant-angle}

\[
\phi(s)
\]

denotes cant rotation.

\end{definition}

% ------------------------------------------------------------

\begin{remark}
Within AIM, \(\phi(s)\) is interpreted geometrically as frame rotation.
Engineering source data may instead describe cant as rail height
difference; the conversion is a convention-dependent runtime or import
operation.
\end{remark}

% ------------------------------------------------------------

\begin{remark}
Depending on context, \(\phi(s)\) may also be interpreted as roll angle
of the local railway frame.
\end{remark}

% ============================================================

\subsection{Cant Evolution}

\begin{definition}[Cant-rate]
\label{def:cant-rate}

\[
\omega(s)
=
\phi'(s)
\]

denotes cant-rate evolution.

\end{definition}

\section{Pose and State Terminology}
% ============================================================

\subsection{Pose}

\begin{definition}[Pose]
\label{def:pose}

A pose denotes a local intrinsic railway state attached to arc-length
position \(s\).

\end{definition}

% ============================================================

\subsection{pose2}

\begin{definition}[pose2]
\label{def:pose2}

The pose2 state is:
\[
\mathrm{pose2}(s)
=
(\gamma(s),\theta(s)).
\]

\end{definition}

% ------------------------------------------------------------

\begin{remark}
pose2 represents intrinsic planar railway geometry.
\end{remark}

% ============================================================

\subsection{pose3}

\begin{definition}[pose3]
\label{def:pose3}

The pose3 state extends pose2 by including spatial orientation and
cant.

A minimal intrinsic representation is:
\[
\mathrm{pose3}(s)
=
(\gamma(s),\theta(s),\phi(s)).
\]

\end{definition}

% ------------------------------------------------------------

\begin{remark}
More elaborate runtime pose3 interpretations may additionally include:
\begin{itemize}
\item local frames,
\item profile evaluation,
\item metric realization,
\item or operational vehicle-space interpretation.
\end{itemize}
\end{remark}

% ============================================================

\subsection{Controls}

\begin{definition}[Control quantities]
\label{def:control-quantities}

The canonical control quantities are:
\[
\kappa(s),
\qquad
\omega(s).
\]

\end{definition}

% ------------------------------------------------------------

\begin{remark}
Curvature controls directional evolution, whereas cant-rate controls
cant evolution.
\end{remark}

% ============================================================

\subsection{State}

\begin{definition}[State]
\label{def:state}

A state denotes a structured railway condition together with its
geometric, operational, or physical interpretation.

\end{definition}

% ------------------------------------------------------------

\begin{remark}
Within AIM, states are not merely stored parameter containers.
\end{remark}

% ============================================================

\subsection{Runtime State}

\begin{definition}[Runtime state]
\label{def:runtime-state}

A runtime state is a dynamically evaluated operational state generated
through runtime operator evaluation.

\end{definition}

% ------------------------------------------------------------

\begin{remark}
Runtime refers to operational evaluation of alignment-related quantities,
independent of implementation details.
\end{remark}

\section{Canonical Terminology}
% ============================================================

\begin{definition}[Preferred terminology]
\label{def:preferred-terminology}

The following terminology is preferred:

\begin{center}
\begin{tabular}{ll}
alignment & intrinsic railway geometry system \\
curve & mathematical geometric object \\
geometry & geometric description \\
trajectory & motion-related path \\
pose & local railway state \\
frame & local spatial interpretation \\
state & structured railway condition \\
runtime state & dynamically evaluated operational state \\
runtime service & operator-based evaluation process \\
realization & physical construction or maintenance process \\
realization operator & mapping from target state to realized state \\
metric realization & metric-dependent spatial interpretation \\
world space & CRS-based spatial coordinates \\
track space & intrinsic alignment coordinates \\
track & physically realized railway structure
\end{tabular}
\end{center}

\end{definition}

% ============================================================
\section{Avoided Ambiguities}
% ============================================================

\begin{remark}
The following ambiguities should be avoided:
\begin{itemize}
\item alignment \(\neq\) merely curve,
\item realization \(\neq\) visualization,
\item runtime \(\neq\) software process only,
\item pose \(\neq\) frame,
\item geometry \(\neq\) realization,
\item target geometry \(\neq\) realized geometry,
\item metric realization \(\neq\) physical realization,
\item coordinate transformation \(\neq\) world-to-track projection,
\item projection \(\neq\) CRS transformation,
\item world coordinates \(\neq\) track coordinates.
\end{itemize}
\end{remark}

% ============================================================
\section{Connection to AIM}
% ============================================================

\begin{summary}

The AIM notation system separates:
\begin{itemize}
\item intrinsic geometry,
\item metric realization,
\item realized geometry,
\item runtime states,
\item operational vehicle-space interpretation,
\item and CRS-dependent world geometry.
\end{itemize}

This notation framework forms the canonical language of AIM.

\end{summary}
