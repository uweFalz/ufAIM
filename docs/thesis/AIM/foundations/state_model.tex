\chapter{State Model}

% ============================================================
\section{Motivation}
% ============================================================

\begin{remark}
Railway alignment modelling involves interacting descriptions of
geometry, orientation, cant, realization, runtime interpretation,
and operational behaviour.
\end{remark}

\begin{remark}
Classical alignment theory often treats these aspects separately:
\begin{itemize}
\item geometry as curves,
\item cant as annotation,
\item realization as construction detail,
\item and optimization as external post-processing.
\end{itemize}
\end{remark}

\begin{remark}
Within AIM, these components are interpreted as parts of a unified
state-based system.
\end{remark}

\begin{remark}
The purpose of the state model is:
\begin{itemize}
\item to define canonical railway states,
\item to distinguish intrinsic and extrinsic representations,
\item to separate target and realized states,
\item and to establish a consistent basis for runtime services,
projection, realization, and optimization.
\end{itemize}
\end{remark}

% ============================================================
\section{State Concept}
% ============================================================

\subsection{State Definition}

\begin{definition}[State]
A state is the minimal set of variables required to describe the
geometric and physical condition of a railway alignment at arc-length
position \(s\).
\end{definition}

\begin{remark}
Within AIM, states are interpreted intrinsically along the alignment
rather than globally in world coordinates.
\end{remark}

\begin{remark}
The alignment is therefore interpreted as a spatial state trajectory
parameterized by arc-length.
\end{remark}

% ------------------------------------------------------------

\subsection{Canonical State Variable}

\begin{definition}[Canonical state]
The canonical intrinsic railway state is:
\begin{equation}
x(s)
=
(\gamma(s),\theta(s),\phi(s)).
\label{eq:state-model-canonical-state}
\end{equation}
\end{definition}

\begin{remark}
The state combines:
\begin{itemize}
\item intrinsic geometry,
\item orientation,
\item and cant evolution.
\end{itemize}
\end{remark}

% ============================================================
\section{Pose2 State}
% ============================================================

\subsection{Definition}

\begin{definition}[Pose2 state]
The planar pose state is:
\begin{equation}
\mathrm{pose2}(s)
=
(\gamma(s),\theta(s)).
\label{eq:pose2-definition}
\end{equation}
\end{definition}

\begin{remark}
The pose2 state describes:
\begin{itemize}
\item planar position,
\item and planar orientation.
\end{itemize}
\end{remark}

% ------------------------------------------------------------

\subsection{Pose2 Evolution}

\begin{remark}
The pose2 evolution is governed by:
\begin{equation}
\theta'(s)
=
\kappa(s),
\label{eq:pose2-curvature}
\end{equation}
together with:
\begin{equation}
\gamma'(s)
=
(\cos\theta(s),\sin\theta(s)).
\label{eq:pose2-position}
\end{equation}
\end{remark}

\begin{remark}
Curvature therefore acts as the primary geometric control variable.
\end{remark}

% ============================================================
\section{Pose3 State}
% ============================================================

\subsection{Definition}

\begin{definition}[Pose3 state]
The extended railway pose state is:
\begin{equation}
\mathrm{pose3}(s)
=
(\gamma(s),\theta(s),\phi(s)).
\label{eq:pose3-definition}
\end{equation}
\end{definition}

\begin{remark}
The pose3 state extends the planar state by including cant.
\end{remark}

\begin{remark}
Cant is interpreted as a rotation around the local tangent direction.
\end{remark}

% ------------------------------------------------------------

\subsection{Controls}

\begin{definition}[Pose3 controls]
The canonical controls are:
\begin{equation}
u(s)
=
(\kappa(s),\omega(s)),
\label{eq:pose3-controls}
\end{equation}
where:
\[
\omega(s)=\phi'(s).
\]
\end{definition}

\begin{remark}
Curvature controls directional evolution, whereas cant-rate controls
cross-level evolution.
\end{remark}

% ------------------------------------------------------------

\subsection{Canonical Pose3 System}

\begin{remark}
The canonical pose3 evolution system is:
\begin{equation}
\begin{aligned}
\gamma'(s)
&=
(\cos\theta(s),\sin\theta(s)),
\\
\theta'(s)
&=
\kappa(s),
\\
\phi'(s)
&=
\omega(s).
\end{aligned}
\label{eq:pose3-evolution}
\end{equation}
\end{remark}

\begin{remark}
This formulation interprets railway alignment as a controlled dynamical
state system.
\end{remark}

% ============================================================
\section{Intrinsic and Extrinsic States}
% ============================================================

\subsection{Intrinsic State}

\begin{definition}[Intrinsic state]
An intrinsic state is defined relative to the alignment itself and uses
arc-length \(s\) as primary parameter.
\end{definition}

\begin{remark}
Examples include:
\begin{itemize}
\item curvature,
\item cant,
\item tangent direction,
\item local railway frames.
\end{itemize}
\end{remark}

% ------------------------------------------------------------

\subsection{Extrinsic State}

\begin{definition}[Extrinsic state]
An extrinsic state is defined relative to a global coordinate reference
system.
\end{definition}

\begin{remark}
Examples include:
\begin{itemize}
\item WGS84 coordinates,
\item projected CRS coordinates,
\item BIM placement coordinates,
\item sensor measurements.
\end{itemize}
\end{remark}

% ------------------------------------------------------------

\subsection{Intrinsic--Extrinsic Coupling}

\begin{remark}
The world--track transformation couples intrinsic and extrinsic states.
\end{remark}

\begin{remark}
World-to-track projection therefore acts as a state conversion process
between global and alignment-relative representations.
\end{remark}

% ============================================================
\section{Target and Realized States}
% ============================================================

\subsection{Target State}

\begin{definition}[Target state]
The target state describes the analytically intended alignment:
\begin{equation}
\mathrm{pose3}_{\mathrm{target}}.
\label{eq:target-state}
\end{equation}
\end{definition}

% ------------------------------------------------------------

\subsection{Realized State}

\begin{definition}[Realized state]
The realized state describes the physically constructed alignment:
\begin{equation}
\mathrm{pose3}_{\mathrm{real}}.
\label{eq:realized-state}
\end{equation}
\end{definition}

% ------------------------------------------------------------

\subsection{Realization Relation}

\begin{remark}
The realization operator connects both states:
\begin{equation}
\mathcal R :
\mathrm{pose3}_{\mathrm{target}}
\mapsto
\mathrm{pose3}_{\mathrm{real}}.
\label{eq:state-realization-relation}
\end{equation}
\end{remark}

\begin{remark}
The realized state includes:
\begin{itemize}
\item construction effects,
\item machine behaviour,
\item discretization,
\item maintenance history,
\item and measurement uncertainty.
\end{itemize}
\end{remark}

% ============================================================
\section{Runtime States}
% ============================================================

\subsection{Runtime Interpretation}

\begin{remark}
Within AIM, states are not static stored objects but runtime-evaluated
entities.
\end{remark}

\begin{remark}
Runtime systems continuously evaluate:
\begin{itemize}
\item world-to-track projections,
\item local frames,
\item realized geometry,
\item dynamic quantities,
\item and optimization residuals.
\end{itemize}
\end{remark}

% ------------------------------------------------------------

\subsection{Derived Runtime Quantities}

\begin{definition}[Derived runtime quantities]
Derived runtime quantities include:
\begin{equation}
a_{\mathrm{eff}},
\qquad
j,
\qquad
\kappa_g,
\qquad
d,
\qquad
h.
\label{eq:runtime-derived-quantities}
\end{equation}
\end{definition}

\begin{remark}
These are not independent state variables, but quantities derived from
the underlying pose and frame structure.
\end{remark}

% ============================================================
\section{Sparse and Hybrid States}
% ============================================================

\subsection{Sparse Representation}

\begin{definition}[Sparse state]
A sparse state is a compact representation using only essential control
parameters.
\end{definition}

\begin{remark}
Examples include:
\begin{itemize}
\item fixed curvature segments,
\item transition segments,
\item cant transitions,
\item anchor poses.
\end{itemize}
\end{remark}

% ------------------------------------------------------------

\subsection{Hybrid Representation}

\begin{definition}[Hybrid state]
A hybrid state combines:
\begin{itemize}
\item parametric structure,
\item and local sampled corrections.
\end{itemize}
\end{definition}

\begin{remark}
Hybrid states combine efficient modelling with local adaptation.
\end{remark}

% ============================================================
\section{Optimization States}
% ============================================================

\subsection{Optimization Variables}

\begin{remark}
Optimization does not necessarily operate on full geometry.
\end{remark}

\begin{remark}
Typical optimization variables include:
\begin{itemize}
\item curvature parameters,
\item transition lengths,
\item cant parameters,
\item control point positions,
\item and realization corrections.
\end{itemize}
\end{remark}

% ------------------------------------------------------------

\subsection{Realization-Aware Optimization}

\begin{remark}
Within AIM, optimization is evaluated on realized states:
\begin{equation}
J(\mathcal R(\mathrm{pose3})).
\label{eq:realization-aware-state-optimization}
\end{equation}
\end{remark}

\begin{remark}
Optimization therefore operates indirectly through the realization
operator.
\end{remark}

% ============================================================
\section{State Hierarchy}
% ============================================================

\begin{remark}
The AIM framework distinguishes:
\begin{itemize}
\item intrinsic states,
\item extrinsic states,
\item target states,
\item realized states,
\item runtime states,
\item and optimization states.
\end{itemize}
\end{remark}

\begin{remark}
These are not independent models, but interconnected layers of one
unified railway state system.
\end{remark}

% ============================================================
\section{Canonical Interpretation}
% ============================================================

\begin{proposition}
Within AIM, railway alignment is interpreted as a controlled spatial
state trajectory evolving along arc-length.
\end{proposition}

\begin{proposition}
Curvature and cant evolution form the primary control structure of this
trajectory.
\end{proposition}

\begin{proposition}
Realization, projection, and optimization operate on states rather than
on isolated geometric entities.
\end{proposition}

% ============================================================
\section{Connection to AIM}
% ============================================================

\begin{summary}
The AIM state model provides a unified framework linking:
\begin{itemize}
\item geometry,
\item cant,
\item realization,
\item runtime projection,
\item dynamics,
\item and optimization.
\end{itemize}

It establishes railway alignment as a coupled state-based system rather
than merely a collection of curves and coordinates.
\end{summary}
