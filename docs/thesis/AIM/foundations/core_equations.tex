\chapter{Core Equations and Canonical Relations}

% ============================================================
\section{Motivation}
% ============================================================

\begin{remark}
The AIM framework is based on a compact set of canonical equations that
connect geometry, dynamics, realization, runtime interpretation, and
optimization.
\end{remark}

\begin{remark}
These equations do not represent isolated formulas, but a coupled state
description for railway alignment modelling.
\end{remark}

\begin{remark}
The purpose of this chapter is not to derive all relations in detail,
but to define the canonical equations repeatedly used throughout AIM.
\end{remark}

\begin{remark}
The equations collected here serve as:
\begin{itemize}
\item canonical reference relations,
\item shared notation anchors,
\item and structural building blocks of the AIM framework.
\end{itemize}
\end{remark}

% ============================================================
\section{Intrinsic Geometry}
% ============================================================

\subsection{Arc-Length State Evolution}

\begin{definition}[State evolution relation]
The intrinsic geometric state evolves according to:
\begin{equation}
\gamma'(s)
=
\mathbf t(s).
\label{eq:state-evolution}
\end{equation}
\end{definition}

\begin{remark}
This relation defines the arc-length parameterization of the alignment.
\end{remark}

% ------------------------------------------------------------

\subsection{Curvature Relation}

\begin{definition}[Curvature relation]
Directional evolution is governed by:
\begin{equation}
\theta'(s)
=
\kappa(s).
\label{eq:curvature-relation}
\end{equation}
\end{definition}

\begin{remark}
Curvature is therefore the differential quantity governing directional
change along the alignment.
\end{remark}

% ------------------------------------------------------------

\subsection{Planar Tangent Representation}

\begin{definition}[Planar tangent relation]
In the planar case:
\begin{equation}
\mathbf t(s)
=
(\cos\theta(s),\sin\theta(s)).
\label{eq:planar-tangent}
\end{equation}
\end{definition}

\begin{remark}
This is a specialization of the intrinsic state evolution relation
\cref{eq:state-evolution}.
\end{remark}

% ------------------------------------------------------------

\subsection{Canonical Pose2 System}

\begin{definition}[Pose2 system]
The canonical planar pose2 system is:
\begin{equation}
\begin{aligned}
\gamma'(s)
&=
\mathbf t(s),
\\
\theta'(s)
&=
\kappa(s).
\end{aligned}
\label{eq:pose2-system}
\end{equation}
\end{definition}

\begin{remark}
Together with \cref{eq:planar-tangent}, this defines the classical
curvature-based planar alignment reconstruction system.
\end{remark}

% ============================================================
\section{Cant Evolution}
% ============================================================

\subsection{Cant State Equation}

\begin{definition}[Cant evolution]
Cant evolution is governed by:
\begin{equation}
\phi'(s)
=
\omega(s),
\label{eq:cant-evolution}
\end{equation}
where \(\omega(s)\) denotes the cant-rate control.
\end{definition}

\begin{remark}
Cant is treated as a state quantity rather than as a secondary
annotation.
\end{remark}

% ------------------------------------------------------------

\subsection{Canonical Pose3 System}

\begin{definition}[Pose3 system]
The canonical pose3 system is:
\begin{equation}
\begin{aligned}
\gamma'(s)
&=
\mathbf t(s),
\\
\theta'(s)
&=
\kappa(s),
\\
\phi'(s)
&=
\omega(s).
\end{aligned}
\label{eq:pose3-system}
\end{equation}
\end{definition}

\begin{remark}
The control variables of the system are:
\[
\kappa(s),
\qquad
\omega(s).
\]
\end{remark}

\begin{remark}
This formulation interprets railway alignment as a controlled dynamical
state system.
\end{remark}

% ------------------------------------------------------------

\subsection{Generalized State Form}

\begin{definition}[Generalized state equation]
The pose evolution may be written abstractly as:
\begin{equation}
x'(s)
=
f(x(s),u(s)),
\label{eq:general-state-system}
\end{equation}
where:
\begin{itemize}
\item \(x(s)\) denotes the railway state,
\item \(u(s)\) denotes the control variables.
\end{itemize}
\end{definition}

\begin{remark}
Within AIM, the controls are primarily:
\[
u(s)
=
(\kappa(s),\omega(s)).
\]
\end{remark}

% ============================================================
\section{Dynamic Relations}
% ============================================================

\subsection{Effective Lateral Acceleration}

\begin{definition}[Effective lateral acceleration]
The effective lateral acceleration is:
\begin{equation}
a_{\mathrm{eff}}
=
v^2\kappa-g\sin\phi.
\label{eq:effective-lateral-acceleration}
\end{equation}
\end{definition}

\begin{remark}
This relation expresses the coupling between curvature and cant.
\end{remark}

% ------------------------------------------------------------

\subsection{Equilibrium Condition}

\begin{definition}[Equilibrium condition]
A perfectly balanced alignment satisfies:
\begin{equation}
v^2\kappa
=
g\sin\phi.
\label{eq:cant-equilibrium}
\end{equation}
\end{definition}

\begin{remark}
This relation defines ideal cant balance.
\end{remark}

% ------------------------------------------------------------

\subsection{Jerk Relation}

\begin{definition}[Jerk relation]
The jerk is:
\begin{equation}
j
=
v^3\kappa'
-
gv\cos\phi\,\phi'.
\label{eq:jerk-relation}
\end{equation}
\end{definition}

\begin{remark}
The jerk depends jointly on curvature evolution and cant evolution.
\end{remark}

% ============================================================
\section{Realization Relations}
% ============================================================

\subsection{Realization Operator}

\begin{definition}[Realization operator]
The realization operator maps target states to realized states:
\begin{equation}
\mathcal R :
\mathrm{pose3}_{\mathrm{target}}
\rightarrow
\mathrm{pose3}_{\mathrm{real}}.
\label{eq:realization-operator}
\end{equation}
\end{definition}

\begin{remark}
Realization includes:
\begin{itemize}
\item construction,
\item machine behaviour,
\item discretization,
\item and structural response.
\end{itemize}
\end{remark}

% ------------------------------------------------------------

\subsection{Curvature Realization Relation}

\begin{definition}[Curvature realization]
At curvature level:
\begin{equation}
\mathcal R(\kappa_{\mathrm{target}})
=
\kappa_{\mathrm{real}}.
\label{eq:curvature-realization}
\end{equation}
\end{definition}

% ------------------------------------------------------------

\subsection{Inverse Realization Problem}

\begin{definition}[Inverse realization]
The inverse realization problem is:
\begin{equation}
\mathcal R(\kappa_{\mathrm{target}})
\approx
\kappa_{\mathrm{desired}}.
\label{eq:inverse-realization}
\end{equation}
\end{definition}

\begin{remark}
Alignment design therefore becomes an inverse problem under realization
constraints.
\end{remark}

% ============================================================
\section{World--Track Transformation}
% ============================================================

\subsection{Track-to-World Mapping}

\begin{definition}[Track-to-world mapping]
The forward transformation is:
\begin{equation}
x(s,d)
=
\gamma(s)+d\,\mathbf n(s).
\label{eq:track-to-world}
\end{equation}
\end{definition}

\begin{remark}
The vector \(\mathbf n(s)\) denotes the local lateral direction.
\end{remark}

% ------------------------------------------------------------

\subsection{World-to-Track Projection}

\begin{definition}[World-to-track projection]
The inverse transformation is:
\begin{equation}
(s^\ast,d^\ast)
=
\arg\min_{s,d}
\|
x-(\gamma(s)+d\,\mathbf n(s))
\|.
\label{eq:world-to-track}
\end{equation}
\end{definition}

\begin{remark}
This is a nonlinear projection problem onto the alignment.
\end{remark}

% ============================================================
\section{Ellipsoid Geometry}
% ============================================================

\subsection{Curvature Decomposition}

\begin{definition}[Curvature decomposition]
The total curvature satisfies:
\begin{equation}
\kappa^2
=
\kappa_g^2+\kappa_n^2.
\label{eq:curvature-decomposition}
\end{equation}
\end{definition}

\begin{remark}
The geodesic curvature \(\kappa_g\) is the physically relevant
curvature for railway dynamics on the Earth surface.
\end{remark}

% ------------------------------------------------------------

\subsection{Intrinsic Surface Curvature}

\begin{definition}[Geodesic curvature]
The intrinsic surface curvature is:
\begin{equation}
\kappa_g
=
\|
\nabla_s\mathbf t
\|.
\label{eq:geodesic-curvature}
\end{equation}
\end{definition}

\begin{remark}
This replaces the purely planar curvature interpretation on curved
reference surfaces.
\end{remark}

% ============================================================
\section{Optimization Relations}
% ============================================================

\subsection{General Optimization Problem}

\begin{definition}[Optimization problem]
A general realization-aware optimization problem is:
\begin{equation}
\min_u
J(\mathcal R(x(u))).
\label{eq:general-optimization}
\end{equation}
\end{definition}

\begin{remark}
The objective is evaluated on realized states rather than purely
analytical states.
\end{remark}

% ------------------------------------------------------------

\subsection{Least-Squares Structure}

\begin{definition}[Least-squares structure]
Many AIM optimization problems take the form:
\begin{equation}
\min_x
\frac12
\|
r(x)
\|^2.
\label{eq:least-squares}
\end{equation}
\end{definition}

% ------------------------------------------------------------

\subsection{Gauss--Newton Approximation}

\begin{definition}[Gauss--Newton approximation]
The Gauss--Newton approximation is:
\begin{equation}
H
\approx
J^T J.
\label{eq:gauss-newton}
\end{equation}
\end{definition}

\begin{remark}
This structure appears naturally in:
\begin{itemize}
\item curvature fitting,
\item realization matching,
\item trajectory optimization,
\item and runtime adjustment problems.
\end{itemize}
\end{remark}

\section{Canonical AIM Interpretation}
% ============================================================

\begin{proposition}
Within AIM, railway alignment is interpreted as a spatial state
trajectory governed by curvature and cant evolution.
\end{proposition}

\begin{proposition}
Curvature, cant, realization, runtime interpretation, and optimization
must therefore be treated as coupled components of one unified system.
\end{proposition}

of AIM.
