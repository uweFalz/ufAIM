% ============================================================
\section*{Motivation}
% ============================================================

\begin{remark}
Classical railway alignment optimization is usually formulated as a
geometric optimization problem.
\end{remark}

\begin{remark}
Typical optimization variables include:
\begin{itemize}
\item points,
\item radii,
\item tangent lengths,
\item and geometric transition parameters.
\end{itemize}
\end{remark}

\begin{remark}
Within AIM, however, operational railway behaviour emerges from runtime
state evolution rather than from geometry alone.
\end{remark}

\begin{remark}
Consequently, the relevant optimization object is not merely geometric
shape, but:
\begin{quote}
runtime operational state trajectories.
\end{quote}
\end{remark}

\begin{principle}[State-space optimization principle]
\label{princ:state-space-optimization}
Within AIM, railway alignment optimization is interpreted fundamentally
as optimization of realized runtime-state trajectories.
\end{principle}

\begin{principle}[Trajectory-shaping principle]
\label{princ:trajectory-shaping}
Railway optimization in AIM is trajectory shaping in runtime state space.
\end{principle}

\section{Optimization Functionals}
% ============================================================

\begin{definition}[Runtime-state functional]
\label{def:runtime-state-functional}
A runtime-state optimization functional may be written as:
\[
J[x]
=
\int_0^L
F(x(s),x'(s),u(s))
\,\dd s.
\]
\end{definition}

\begin{remark}
The functional \(J[x]\) evaluates the quality of an operational
runtime-state trajectory.
\end{remark}

\begin{remark}
Typical optimization targets include:
\begin{itemize}
\item minimizing jerk,
\item minimizing effective acceleration variation,
\item minimizing roll excitation,
\item minimizing realization sensitivity,
\item satisfying operational envelopes,
\item and maximizing operational smoothness.
\end{itemize}
\end{remark}

\begin{remark}
This shifts optimization away from isolated geometric residuals such as
radius errors, tangent errors, or point-list errors toward functional
evaluation of runtime-state trajectories.
\end{remark}
