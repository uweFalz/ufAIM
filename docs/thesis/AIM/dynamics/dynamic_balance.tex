\chapter{Dynamic Balance}

\label{chap:dynamic-balance}

% ============================================================
\section{Motivation}
% ============================================================

\begin{remark}
Railway operation continuously couples:
\begin{itemize}
\item curvature,
\item cant,
\item and operational velocity.
\end{itemize}
\end{remark}

\begin{remark}
Classical railway engineering expresses this interaction through
equilibrium relations between curvature-induced acceleration and cant.
\end{remark}

\begin{remark}
Within AIM, these relations are interpreted as runtime-state coupling
laws rather than isolated engineering formulas.
\end{remark}

\begin{principle}[Dynamic-balance principle]
\label{princ:dynamic-balance}
Operational balance emerges from runtime coupling between geometry,
cant, and operational motion.
\end{principle}

% ============================================================
\section{Balance Relation}
% ============================================================

\begin{definition}[Balance relation]
\label{def:balance-relation}
A dynamically balanced state approximately satisfies:
\[
v^2\kappa(s)
=
g\sin\phi(s).
\]
\end{definition}

\begin{remark}
The balance relation couples:
\begin{itemize}
\item curvature-induced acceleration,
\item and cant-induced compensation.
\end{itemize}
\end{remark}

\begin{remark}
Within AIM, this relation is interpreted as a runtime-state coupling law.
\end{remark}

% ============================================================
\section{Operational Imbalance}
% ============================================================

\begin{definition}[Operational imbalance]
\label{def:operational-imbalance}
Operational imbalance is the deviation from the balance relation.
\end{definition}

\begin{definition}[Effective lateral acceleration]
\label{def:dynamics-effective-lateral-acceleration}
The canonical imbalance quantity is:
\[
a_{\mathrm{eff}}
=
v^2\kappa
-
g\sin\phi.
\]
\end{definition}

\begin{remark}
The effective acceleration measures the residual imbalance between
curvature and cant.
\end{remark}

\begin{remark}
Perfect balance is rarely achieved in practical railway operation.
\end{remark}

% ============================================================
\section{Cant Deficiency and Excess Cant}
% ============================================================

\begin{definition}[Cant deficiency]
\label{def:cant-deficiency}
Cant deficiency occurs whenever:
\[
v^2\kappa
>
g\sin\phi.
\]
\end{definition}

\begin{remark}
In this case, uncompensated lateral acceleration remains.
\end{remark}

\begin{definition}[Excess cant]
\label{def:excess-cant}
Excess cant occurs whenever:
\[
v^2\kappa
<
g\sin\phi.
\]
\end{definition}

\begin{remark}
Both situations correspond to operational imbalance states.
\end{remark}

% ============================================================
\section{Balance Trajectories}
% ============================================================

\begin{definition}[Balance trajectory]
\label{def:dynamics-balance-trajectory}
A balance trajectory is the evolution of balance states along the
alignment.
\end{definition}

\begin{remark}
Balance trajectories are generated jointly by:
\begin{itemize}
\item curvature evolution,
\item cant evolution,
\item and operational velocity.
\end{itemize}
\end{remark}

\begin{remark}
Operational smoothness therefore depends on the evolution of balance
states rather than on isolated equilibrium conditions.
\end{remark}

% ============================================================
\section{Runtime-State Coupling}
% ============================================================

\begin{remark}
The balance relation:
\[
v^2\kappa
=
g\sin\phi
\]
connects:
\begin{itemize}
\item intrinsic geometry,
\item cant,
\item runtime dynamics,
\item and operational behaviour.
\end{itemize}
\end{remark}

\begin{remark}
Dynamic balance therefore belongs naturally to runtime-state evolution.
\end{remark}

% ============================================================
\section{Canonical Interpretation}
% ============================================================

\begin{proposition}
Within AIM, dynamic balance is interpreted as runtime-state coupling.
\end{proposition}

\begin{proposition}
Operational imbalance is represented by effective lateral acceleration.
\end{proposition}

\begin{proposition}
Cant deficiency and excess cant are imbalance states.
\end{proposition}

\begin{proposition}
Balance trajectories describe the evolution of operational balance along
the alignment.
\end{proposition}

% ============================================================
\section{Connection to AIM}
% ============================================================

\begin{summary}
Dynamic balance provides the fundamental coupling between:
\begin{itemize}
\item curvature,
\item cant,
\item and operational velocity.
\end{itemize}

Within AIM:
\begin{itemize}
\item balance is interpreted as a runtime-state relation,
\item effective acceleration measures imbalance,
\item cant deficiency and excess cant become imbalance states,
\item and operational behaviour is described through balance
trajectories.
\end{itemize}

Dynamic balance therefore forms the bridge between intrinsic geometry
and operational runtime behaviour.
\end{summary}
