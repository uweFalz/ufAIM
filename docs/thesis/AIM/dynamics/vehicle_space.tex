\chapter{Vehicle Space and Relative Railway States}
\label{chap:vehicle-space}

\section{Introduction}

This chapter establishes how vehicle-related operational quantities are
interpreted relative to the railway runtime state. Within AIM, the
canonical railway state remains \(\mathrm{pose3}\), while vehicle-space
quantities are derived interpretation states used for operation,
assessment, and control support.

\begin{principle}[Track--Vehicle Distinction]
Within AIM, railway states and vehicle states are fundamentally distinct.
Vehicle states are interpreted relative to railway states.
\end{principle}

\section{Track State versus Vehicle State}

\begin{proposition}[State hierarchy]
\label{prop:state-hierarchy}

Railway states exist independently of vehicles.

Vehicle states require railway states as reference states.

Therefore:

\[
\mathrm{vehicleState}
=
\mathcal V(\mathrm{pose3},\ldots)
\]

but not vice versa.

\end{proposition}

\begin{definition}[Track State]
The track state is the runtime-evaluated railway state represented by
pose3 and its associated railway reference frame.
\end{definition}

\begin{definition}[Vehicle State]
A vehicle state is an operational state interpreted relative to a track
state.
\end{definition}

\begin{equation}
\mathrm{vehicleState}
=
\mathcal V
(
\mathrm{pose3},
\dots
)
\end{equation}

\section{Vehicle Space}

\begin{definition}[Vehicle Space]
Vehicle space denotes the collection of operational states that are
interpreted relative to a railway state.
\end{definition}

Vehicle space therefore provides a structured application layer for
clearance, comfort, and dynamic interpretation without redefining
canonical railway identity.

\section{Relative Vehicle States}

\begin{definition}[Relative Vehicle State]
A relative vehicle state is a vehicle state expressed in a local
track-attached reference frame.
\end{definition}

Representative relative states include:

\begin{itemize}
\item lateral displacement
\item yaw deviation
\item roll deviation
\item wheel unloading
\end{itemize}

\section{Track and Vehicle Frames}

\begin{definition}[Track Frame]
The track frame is a local railway-attached frame generated from
\(\mathrm{pose3}\).
\end{definition}

\begin{definition}[Vehicle Frame]
The vehicle frame is a local frame attached to a vehicle body or
subsystem.
\end{definition}

\begin{equation}
F_{\mathrm{vehicle}}
\neq
F_{\mathrm{track}}
\end{equation}

This frame distinction is the basis for expressing relative motion and
evaluating operational criteria in track-attached coordinates.

\section{Operational Abstractions}

\subsection{Wheelset State}

\begin{definition}
The wheelset state denotes operational quantities associated with a
wheelset relative to the local track frame.
\end{definition}

\subsection{Bogie State}

\begin{definition}
The bogie state denotes operational quantities associated with a bogie
assembly relative to the local track frame.
\end{definition}

\subsection{Center-of-Gravity State}

\begin{definition}
The center-of-gravity state denotes the motion state of a selected
vehicle mass reference point relative to the local track frame.
\end{definition}


The center-of-gravity state is not identical to the vehicle state.

A vehicle may contain multiple relevant operational states,
whereas the center-of-gravity state describes the motion of a selected
mass reference point.



Within AIM, future optimization approaches may operate on
center-of-gravity trajectories rather than directly on geometric
centerlines.


\section{Canonical Interpretation}

\begin{proposition}
pose3 describes the railway rather than the vehicle.
\end{proposition}

\begin{proposition}
Vehicle states are relative interpretation states.
\end{proposition}

\begin{proposition}
Vehicle space is generated from railway states.
\end{proposition}

\section{Summary}

Vehicle space introduces the operational entities that are interpreted
relative to railway states.

It defines the ontology of vehicle-related states but does not yet
describe runtime evolution, dynamic balance, realization effects, or
optimization.
