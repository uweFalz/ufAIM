% ============================================================
\section{Motivation}
% ============================================================

\begin{remark}
Classical railway alignment theory often assumes that operational
dynamics arise directly from analytical target geometry.
\end{remark}

\begin{remark}
In reality, however, railway operation interacts with:
\begin{itemize}
\item realized geometry,
\item measured states,
\item runtime reconstruction,
\item and realization-dependent operational behaviour.
\end{itemize}
\end{remark}

\begin{remark}
Thus, operational railway dynamics are fundamentally realization-aware.
\end{remark}

\begin{remark}
Within AIM, runtime dynamics are therefore interpreted not as direct
consequences of analytical geometry, but as consequences of realized and
runtime-evaluated railway states.
\end{remark}

\begin{principle}[Realization-aware dynamics principle]
\label{princ:realization-aware-dynamics}
Within AIM, operational railway dynamics emerge from realized runtime
states rather than from idealized analytical geometry alone.
\end{principle}
