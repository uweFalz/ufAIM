\chapter{Realization-Aware Runtime Dynamics}

\label{chap:realization-aware-dynamics}

% ============================================================
\section{Motivation}
% ============================================================

\begin{remark}
Classical railway alignment theory often assumes that operational
dynamics arise directly from analytical target geometry.
\end{remark}

\begin{remark}
Within AIM, this assumption is insufficient.
Operational railway behaviour depends on realized and runtime-evaluated
railway states.
\end{remark}

\begin{remark}
The relevant chain is:
\[
\mathrm{target}
\rightarrow
\mathrm{realization}
\rightarrow
\mathrm{runtime}
\rightarrow
\mathrm{operation}.
\]
\end{remark}

\begin{principle}[Realization-aware dynamics principle]
\label{princ:realization-aware-dynamics}
Operational railway dynamics emerge from realized runtime states rather
than from idealized analytical geometry alone.
\end{principle}

% ============================================================
\section{Realized Runtime States}
% ============================================================

\begin{definition}[Realization-aware runtime state]
\label{def:realization-aware-runtime-state}
A realization-aware runtime state is a runtime railway state evaluated
from realized rather than purely analytical geometry.
\end{definition}

\begin{remark}
Within AIM, the operationally relevant railway state is:
\[
\mathrm{pose3}_{\mathrm{real}}(s).
\]
\end{remark}

\begin{remark}
This state includes realized geometry, realized cant, realized curvature,
runtime reconstruction, and operational interpretation.
\end{remark}

\begin{remark}
The analytical target state remains important, but only through the
realized runtime state that it generates.
\end{remark}

% ============================================================
\section{Realization Operator}
% ============================================================

\begin{definition}[Pose3 realization operator]
\label{def:pose3-realization-operator-runtime}
The realization operator acting on pose3 states is:
\[
\mathcal R_{\mathrm{pose3}}
:
\mathrm{pose3}_{\mathrm{target}}
\mapsto
\mathrm{pose3}_{\mathrm{real}}.
\]
\end{definition}

\begin{remark}
The operator \(\mathcal R_{\mathrm{pose3}}\) includes construction
effects, measurement effects, realization filtering, maintenance
history, mechanical response, and runtime reconstruction.
\end{remark}

\begin{remark}
Operational runtime dynamics therefore depend on:
\[
\mathcal D
\bigl(
\mathcal R_{\mathrm{pose3}}
(
\mathrm{pose3}_{\mathrm{target}}
)
\bigr).
\]
\end{remark}

\begin{principle}[Realization-operator principle]
\label{princ:realization-operator-principle}
Within AIM, realization is a central runtime-state operator, not a
post-processing correction.
\end{principle}

% ============================================================
\section{Measurement and Reconstruction}
% ============================================================

\begin{remark}
Operational railway systems do not access exact analytical geometry
directly.
\end{remark}

\begin{remark}
Instead, runtime systems operate on measurements, reconstructed states,
estimated geometry, and realization-aware runtime evaluation.
\end{remark}

\begin{remark}
Measured states are evidence for realized runtime states.
They are not automatically identical to the realized state itself.
\end{remark}

% ============================================================
\section{Realized Dynamic States}
% ============================================================

\begin{definition}[Realized dynamic state]
\label{def:realized-dynamic-state}
A realized dynamic state is an operational runtime state generated from:
\[
\mathrm{pose3}_{\mathrm{real}}(s).
\]
\end{definition}

\begin{remark}
Important realized dynamic quantities include:
\begin{itemize}
\item effective acceleration,
\item jerk,
\item roll evolution,
\item cant deficiency,
\item balance trajectories,
\item and center-of-gravity motion.
\end{itemize}
\end{remark}

\begin{remark}
These quantities may differ from their analytical target counterparts
because realization changes the runtime-generating state.
\end{remark}

% ============================================================
\section{Realized Effective Acceleration}
% ============================================================

\begin{definition}[Realized effective acceleration]
\label{def:realized-effective-acceleration-runtime}
The realized effective acceleration is:
\[
a_{\mathrm{eff,real}}(s)
=
v^2\kappa_{\mathrm{real}}(s)
-
g\sin\phi_{\mathrm{real}}(s).
\]
\end{definition}

\begin{remark}
This quantity is operationally more relevant than the corresponding
target quantity:
\[
a_{\mathrm{eff,target}}(s).
\]
\end{remark}

\begin{remark}
Passenger comfort and admissibility depend on realized rather than
idealized runtime states.
\end{remark}

% ============================================================
\section{Realized Jerk}
% ============================================================

\begin{definition}[Realized jerk]
\label{def:realized-runtime-jerk}
The realized jerk is:
\[
j_{\mathrm{real}}(s)
=
v^3\kappa'_{\mathrm{real}}(s)
-
gv\cos\phi_{\mathrm{real}}(s)\,
\phi'_{\mathrm{real}}(s).
\]
\end{definition}

\begin{remark}
Realized jerk depends jointly on realized curvature evolution, realized
cant evolution, and runtime interpretation.
\end{remark}

\begin{remark}
Thus, comfort depends fundamentally on realization-aware runtime
behaviour.
\end{remark}

% ============================================================
\section{Realization Filtering}
% ============================================================

\begin{remark}
Realization acts approximately as a low-pass filter on curvature, cant,
and higher-order runtime-state evolution.
\end{remark}

\begin{remark}
High-frequency operational behaviour may be attenuated during
construction, maintenance, reconstruction, and runtime evaluation.
\end{remark}

\begin{proposition}[Runtime realization-filter principle]
\label{prop:runtime-realization-filter}
Operational runtime behaviour is generated from filtered realized states
rather than directly from analytical target geometry.
\end{proposition}

\begin{remark}
A transition law is operationally meaningful only if its intended
runtime behaviour survives realization.
\end{remark}

% ============================================================
\section{Realization-Aware Admissibility}
% ============================================================

\begin{definition}[Realization-aware admissibility]
\label{def:realization-aware-admissibility}
A railway state is realization-aware admissible if its realized runtime
states remain within the relevant operational envelope.
\end{definition}

\begin{remark}
Admissibility depends jointly on geometry, realization, runtime
reconstruction, operational interpretation, and project-specific
constraints.
\end{remark}

\begin{remark}
Operational admissibility is therefore not merely a design-time check.
It is a realization-aware runtime-state property.
\end{remark}

% ============================================================
\section{Runtime Envelopes}
% ============================================================

\begin{definition}[Realization-aware runtime envelope]
\label{def:realization-runtime-envelope}
A realization-aware runtime envelope is the admissible region of
realized operational runtime states.
\end{definition}

\begin{remark}
Such envelopes may constrain realized acceleration, realized jerk, roll
evolution, cant deficiency, dynamic balance, and operational smoothness.
\end{remark}

\begin{remark}
Runtime envelopes provide the constraint domain for realization-aware
optimization.
\end{remark}

reconstruction, operational dynamics, admissibility, and optimization.

\begin{itemize}
\item the realization operator acts on runtime-state generation,
\item operational behaviour emerges from realized geometry,
\item effective acceleration and jerk become realized quantities,

runtime dynamics, Schwerpunkt-Trassierung, engineering admissibility, and
