\chapter{Runtime Dynamics}

\label{chap:runtime-dynamics}

% ============================================================
\section{Motivation}
% ============================================================

\begin{remark}
Railway alignments are not merely geometric structures.
\end{remark}

\begin{remark}
During operation, railway geometry generates evolving operational states
through interaction with moving vehicles.
\end{remark}

\begin{remark}
Within AIM, these operational quantities are interpreted as runtime
states derived from:
\begin{itemize}
\item intrinsic curvature evolution,
\item cant evolution,
\item runtime frame evaluation,
\item realization,
\item and vehicle-space interpretation.
\end{itemize}
\end{remark}

\begin{remark}
This motivates interpretation of railway dynamics as runtime state
evolution rather than merely as static geometry.
\end{remark}

\begin{principle}[Runtime-dynamics principle]
\label{princ:runtime-dynamics}
Within AIM, operational railway dynamics are interpreted as runtime
evolution of coupled railway and vehicle states.
\end{principle}

% ============================================================
\section{Runtime State Evolution}
% ============================================================

\begin{definition}[Runtime state]
\label{def:runtime-state}
A runtime state is an operationally evaluated state depending on:
\begin{itemize}
\item pose reconstruction,
\item frame evaluation,
\item realization,
\item operational context,
\item and vehicle interpretation.
\end{itemize}
\end{definition}

% ------------------------------------------------------------

\begin{definition}[Runtime state evolution]
\label{def:runtime-state-evolution}
Runtime evolution may formally be written as:
\[
x'(s)
=
f(x,u),
\]
where:
\begin{itemize}
\item \(x(s)\) denotes the runtime state,
\item \(u(s)\) denotes operational or geometric input,
\item and \(f\) denotes the runtime evolution operator.
\end{itemize}
\end{definition}

% ------------------------------------------------------------

\begin{remark}
Within AIM, the evolution parameter is typically arc-length rather than
time.
\end{remark}

% ------------------------------------------------------------

\begin{remark}
The runtime state may include:
\begin{itemize}
\item curvature,
\item cant,
\item roll state,
\item effective acceleration,
\item vehicle-relative quantities,
\item and operational frame states.
\end{itemize}
\end{remark}

% ============================================================
\section{Runtime Dynamics Operator}
% ============================================================

\begin{definition}[Runtime dynamics operator]
\label{def:runtime-dynamics-operator}
The runtime dynamics operator is denoted by:
\[
\mathcal D.
\]
\end{definition}

% ------------------------------------------------------------

\begin{remark}
Operational runtime evolution may therefore be interpreted abstractly
as:
\[
\mathcal D(\mathrm{pose3}).
\]
\end{remark}

% ------------------------------------------------------------

\begin{remark}
The operator \(\mathcal D\) does not represent a single mechanical
model.
Instead, it denotes a runtime interpretation layer generating
operational state evolution from railway geometry.
\end{remark}

% ============================================================
\section{Acceleration Interpretation}
% ============================================================

\begin{definition}[Runtime acceleration]
\label{def:runtime-acceleration}
Runtime acceleration denotes operational acceleration quantities derived
from runtime state evaluation.
\end{definition}

% ------------------------------------------------------------

\begin{remark}
Important quantities include:
\begin{itemize}
\item lateral acceleration,
\item effective lateral acceleration,
\item roll-induced acceleration,
\item and vertical acceleration components.
\end{itemize}
\end{remark}

% ------------------------------------------------------------

\begin{remark}
Within AIM, acceleration is interpreted operationally relative to the
runtime railway frame rather than as a purely world-space quantity.
\end{remark}

% ============================================================
\section{Effective Lateral Acceleration}
% ============================================================

\begin{definition}[Effective lateral acceleration]
\label{def:runtime-effective-acceleration}
The effective lateral acceleration is:
\[
a_{\mathrm{eff}}(s)
=
v^2\kappa(s)
-
g\sin\phi(s).
\]
\end{definition}

% ------------------------------------------------------------

\begin{remark}
This quantity describes the operational imbalance between:
\begin{itemize}
\item curvature-induced acceleration,
\item and cant-induced compensation.
\end{itemize}
\end{remark}

% ------------------------------------------------------------

\begin{remark}
The effective acceleration is one of the primary runtime quantities
governing:
\begin{itemize}
\item passenger comfort,
\item load transfer,
\item and operational admissibility.
\end{itemize}
\end{remark}

% ============================================================
\section{Cant Deficiency}
% ============================================================

\begin{definition}[Cant deficiency]
\label{def:cant-deficiency}
Cant deficiency denotes the operational mismatch between:
\begin{itemize}
\item required equilibrium cant,
\item and realized or applied cant.
\end{itemize}
\end{definition}

% ------------------------------------------------------------

\begin{remark}
Cant deficiency is therefore a runtime operational quantity rather than
a purely geometric parameter.
\end{remark}

% ------------------------------------------------------------

\begin{remark}
Within AIM, cant deficiency depends jointly on:
\begin{itemize}
\item curvature,
\item cant,
\item speed,
\item realization,
\item and runtime interpretation.
\end{itemize}
\end{remark}

% ============================================================
\section{Jerk Interpretation}
% ============================================================

\begin{definition}[Runtime jerk]
\label{def:runtime-jerk}
Runtime jerk describes the evolution of acceleration along the runtime
trajectory.
\end{definition}

% ------------------------------------------------------------

\begin{remark}
For railway operation, jerk depends jointly on:
\[
\kappa'(s),
\qquad
\phi'(s),
\qquad
v.
\]
\end{remark}

% ------------------------------------------------------------

\begin{remark}
One canonical relation is:
\[
j(s)
=
v^3\kappa'(s)
-
gv\cos\phi(s)\,\phi'(s).
\]
\end{remark}

% ------------------------------------------------------------

\begin{remark}
Jerk is one of the central runtime indicators for:
\begin{itemize}
\item comfort,
\item transition quality,
\item and dynamic smoothness.
\end{itemize}
\end{remark}

% ============================================================
\section{Roll Response}
% ============================================================

\begin{definition}[Roll response]
\label{def:roll-response}
Roll response describes the operational rotational response of the
vehicle state relative to the track frame.
\end{definition}

% ------------------------------------------------------------

\begin{remark}
Roll response depends on:
\begin{itemize}
\item cant evolution,
\item suspension behaviour,
\item center-of-gravity position,
\item and operational velocity.
\end{itemize}
\end{remark}

% ------------------------------------------------------------

\begin{remark}
Within AIM, the roll state is interpreted relative to the runtime track
frame generated by pose3 evaluation.
\end{remark}

% ============================================================
\section{Dynamic Coupling}
% ============================================================

\begin{remark}
Operational railway quantities are strongly coupled.
\end{remark}

% ------------------------------------------------------------

\begin{remark}
In particular:
\begin{itemize}
\item curvature influences acceleration,
\item cant influences roll response,
\item realization influences smoothness,
\item and runtime interpretation influences operational admissibility.
\end{itemize}
\end{remark}

% ------------------------------------------------------------

\begin{remark}
Thus, railway dynamics cannot be interpreted as isolated scalar
quantities.
\end{remark}

% ------------------------------------------------------------

\begin{proposition}[Dynamic coupling principle]
\label{prop:dynamic-coupling}
Operational railway behaviour emerges from coupled runtime evolution of:
\begin{itemize}
\item geometry,
\item cant,
\item realization,
\item runtime frames,
\item and vehicle-space interpretation.
\end{itemize}
\end{proposition}

% ============================================================
\section{Runtime Dynamics versus Full Mechanics}
% ============================================================

\begin{remark}
The AIM runtime-dynamics framework is not intended as a complete
multibody simulation model.
\end{remark}

% ------------------------------------------------------------

\begin{remark}
Instead, it provides:
\begin{itemize}
\item operational runtime interpretation,
\item state coupling,
\item admissibility evaluation,
\item and realization-aware state evolution.
\end{itemize}
\end{remark}

% ------------------------------------------------------------

\begin{remark}
Detailed vehicle mechanics may later be coupled to this framework
through specialized dynamic models.
\end{remark}

\section{Connection to AIM}
% ============================================================

\begin{summary}
Runtime dynamics extend AIM geometry toward operational railway-state
evolution.

Within this framework:
\begin{itemize}
\item runtime states evolve along railway geometry,
\item acceleration and jerk become runtime quantities,
\item cant deficiency becomes an operational state mismatch,
\item realization affects dynamic behaviour directly,
\item and curvature evolution generates operational dynamics.
\end{itemize}

This establishes the bridge between:
\begin{itemize}
\item intrinsic geometry,
\item runtime evaluation,
\item vehicle-space interpretation,
\item realization,
\item and operational state design.
\end{itemize}
\end{summary}
