\chapter{Runtime Dynamics}

\label{chap:runtime-dynamics}

% ============================================================
\section{Motivation}
% ============================================================

\begin{remark}
Railway alignments are not merely geometric objects.
\end{remark}

\begin{remark}
During operation, railway geometry generates evolving operational
states through interaction with runtime evaluation, realization, and
vehicle interpretation.
\end{remark}

\begin{remark}
Within AIM, these operational quantities are not treated as persistent
model parameters.
They are interpreted as runtime-derived state trajectories.
\end{remark}

\begin{principle}[Runtime-dynamics principle]
\label{princ:runtime-dynamics}
Within AIM, operational railway dynamics are interpreted as runtime
state evolution generated from railway runtime states.
\end{principle}

% ============================================================
\section{Dynamic Runtime States}
% ============================================================

\begin{definition}[Dynamic runtime state]
\label{def:runtime-dynamic-state}
A dynamic runtime state is an operational state that evolves during
runtime interpretation.
\end{definition}

\begin{remark}
Dynamic runtime states are generated from:
\begin{itemize}
\item railway runtime states,
\item operational context,
\item realization,
\item and runtime evaluation services.
\end{itemize}
\end{remark}

\begin{remark}
They do not belong to the intrinsic alignment representation.
Instead, they arise from interpretation of alignment behaviour.
\end{remark}

\begin{remark}
Examples include:
\begin{itemize}
\item acceleration-related states,
\item comfort-related states,
\item admissibility-related states,
\item balance-related states,
\item and vehicle-relative operational states.
\end{itemize}
\end{remark}

% ============================================================
\section{Runtime Dynamics Operator}
% ============================================================

\begin{definition}[Runtime dynamics operator]
\label{def:runtime-dynamics-operator}
The runtime dynamics operator is denoted by:
\[
\mathcal D.
\]
\end{definition}

\begin{remark}
The operator
\[
\mathcal D
\]
maps railway runtime states to dynamic operational states.
\end{remark}

\begin{remark}
Conceptually:
\[
x(s)
=
\mathcal D
\bigl(
\mathrm{pose3}(s),
\mathcal C
\bigr),
\]
where:
\begin{itemize}
\item \(\mathrm{pose3}(s)\) denotes the railway runtime state,
\item \(\mathcal C\) denotes operational context,
\item and \(x(s)\) denotes a dynamic runtime state.
\end{itemize}
\end{remark}

\begin{remark}
The operator
\[
\mathcal D
\]
does not represent a specific physical vehicle model.
It represents the dynamic interpretation layer of AIM.
\end{remark}

\section{Arc-Length State Evolution}
% ============================================================

\begin{definition}[Runtime state evolution]
\label{def:runtime-dynamics-state-evolution}
Runtime state evolution may be represented abstractly by:
\[
x'(s)
=
f(x(s),u(s)),
\]
where:
\begin{itemize}
\item \(x(s)\) denotes the dynamic state,
\item \(u(s)\) denotes operational input,
\item and \(f\) denotes the evolution law.
\end{itemize}
\end{definition}

\begin{remark}
Within AIM, state evolution is naturally formulated over arc-length
rather than time.
\end{remark}

\begin{remark}
The intrinsic parameter
\[
s
\]
is already present in alignment geometry and therefore provides a
natural evolution coordinate.
\end{remark}

\begin{remark}
For known velocity
\[
v(s),
\]
time-based and arc-length-based derivatives satisfy:
\[
\frac{\dd}{\dd t}
=
v
\frac{\dd}{\dd s}.
\]
\end{remark}

\begin{remark}
This relation provides the bridge between geometric description and
operational interpretation.
\end{remark}

% ============================================================
\section{Runtime Dynamics versus Geometry}
% ============================================================

\begin{remark}
Runtime dynamics should not be confused with intrinsic geometry.
\end{remark}

\begin{remark}
Intrinsic geometry describes:
\begin{itemize}
\item curvature,
\item profile,
\item cant,
\item and alignment reconstruction.
\end{itemize}
\end{remark}

\begin{remark}
Runtime dynamics describe the operational consequences generated from
these quantities.
\end{remark}

\begin{remark}
Geometry defines structure.

Runtime dynamics describe behaviour.
\end{remark}

\begin{proposition}
\label{prop:dynamics-not-geometry}
Dynamic runtime states are derived from geometry but are not themselves
geometric quantities.
\end{proposition}

% ============================================================
\section{Runtime Dynamics versus Mechanics}
% ============================================================

\begin{remark}
The AIM runtime-dynamics framework is not intended as a complete
multibody simulation framework.
\end{remark}

\begin{remark}
Instead, it provides:
\begin{itemize}
\item operational interpretation,
\item state evolution,
\item admissibility evaluation,
\item and dynamic state generation.
\end{itemize}
\end{remark}

\begin{remark}
Detailed vehicle mechanics may be connected to AIM through dedicated
simulation models.
\end{remark}

\begin{principle}[Mechanics separation principle]
\label{princ:mechanics-separation}
Runtime dynamics in AIM form an operational state-evolution layer rather
than a complete vehicle-dynamics simulation layer.
\end{principle}

% ============================================================
\section{Runtime Dynamics and Curvature Space}
% ============================================================

\begin{remark}
Runtime dynamics originate from curvature evolution.
\end{remark}

\begin{remark}
The quantities
\[
\kappa(s),
\qquad
\kappa'(s),
\qquad
\kappa''(s)
\]
are therefore not only geometric descriptors.
They are generators of dynamic runtime behaviour.
\end{remark}

\begin{remark}
Curvature space may thus be interpreted as a state-generation domain.
\end{remark}

\begin{remark}
This observation forms one of the central motivations of AIM:
operational behaviour emerges from intrinsic curvature structure.
\end{remark}

% ============================================================
\section{Canonical Interpretation}
% ============================================================

\begin{proposition}
Within AIM, railway dynamics are interpreted as runtime state evolution.
\end{proposition}

\begin{proposition}
Dynamic runtime states are generated from railway runtime states.
\end{proposition}

\begin{proposition}
The runtime dynamics operator
\[
\mathcal D
\]
maps railway runtime states to operational state trajectories.
\end{proposition}

\begin{proposition}
Runtime dynamics belong to the operational interpretation layer and not
to intrinsic geometry itself.
\end{proposition}

\begin{proposition}
The AIM runtime-dynamics framework provides operational state evolution
rather than complete vehicle mechanics.
\end{proposition}

% ============================================================
\section{Connection to AIM}
% ============================================================

\begin{summary}
Runtime dynamics extend AIM from geometric reconstruction toward
operational state evolution.

Within this framework:
\begin{itemize}
\item pose3 provides railway runtime states,
\item the operator \(\mathcal D\) generates dynamic runtime states,
\item state evolution is naturally formulated over arc-length,
\item curvature evolution acts as a generator of runtime behaviour,
\item geometry and dynamics remain conceptually separated,
\item and operational interpretation remains separated from detailed
vehicle mechanics.
\end{itemize}

Runtime dynamics therefore establish the bridge between:
\begin{itemize}
\item intrinsic geometry,
\item runtime evaluation,
\item operational interpretation,
\item and later realization-aware optimization.
\end{itemize}
\end{summary}
