\chapter{Railway Alignment}

\section{Motivation}

\begin{remark}
Railway alignment represents one of the most demanding application
domains for curvature-based geometric modelling.
\end{remark}

\begin{remark}
Unlike generic curve design, railway alignments are subject to strict
constraints involving curvature, transition behaviour, cant,
longitudinal profile, operational speed, and safety regulations.
\end{remark}

\begin{remark}
This makes railway alignment a natural test case for any alignment-based
modelling framework.
\end{remark}

---

\section{Core Components of Railway Geometry}

\begin{definition}[Track alignment]
A track alignment is the spatial path followed by a railway track,
typically described in terms of horizontal alignment, vertical profile,
and cant.
\end{definition}

\subsection{Horizontal Alignment}

\begin{remark}
The horizontal alignment is the planar projection of the track geometry.
It is primarily governed by curvature.
\end{remark}

\begin{itemize}
\item straight segments ($\kappaf = 0$),
\item circular arcs ($\kappaf = const$),
\item transition elements (varying $\kappaf$).
\end{itemize}

\subsection{Vertical Profile}

\begin{remark}
The vertical profile describes elevation as a function of station.
It introduces additional geometric structure, often modelled
independently from horizontal curvature.
\end{remark}

\OPEN{Unified treatment of horizontal and vertical geometry.}

\subsection{Cant}

\begin{definition}[Cant]
Cant is the elevation difference between the rails, used to compensate
lateral acceleration in curved track segments.
\end{definition}

\begin{remark}
Cant introduces an additional functional layer that interacts with
curvature and speed.
\end{remark}

\section{Coupling of Parameters}

\begin{remark}
Railway alignment is characterized by strong coupling between multiple
parameters:
\end{remark}

\begin{itemize}
\item curvature $\kappaf$,
\item cant,
\item speed,
\item transition length.
\end{itemize}

\begin{remark}
These parameters are not independent. For example:
\begin{itemize}
\item higher speed requires lower curvature or higher cant,
\item cant variation imposes constraints on transition behaviour,
\item transition length is linked to comfort and safety requirements.
\end{itemize}
\end{remark}

\section{Engineering Constraints}

Railway alignments are governed by engineering rules, including:
\end{remark}

\begin{itemize}
\item minimum radius,
\item maximum cant,
\item maximum cant gradient,
\item limits on curvature derivatives.
\end{itemize}

\begin{remark}

\section{Why Railway Alignment is a Key Use Case}

\begin{remark}
Railway alignment combines:
\begin{itemize}
\item strict geometric constraints,
\item strong parameter coupling,
\item real-world data imperfections,
\item and high safety requirements.
\end{itemize}
\end{remark}

\begin{remark}
This makes it an ideal domain for testing and validating the concepts
introduced in this manuscript.
\end{remark}

\section{Summary}

\begin{summary}
Railway alignment is a highly structured and constraint-driven domain
for geometric modelling.

It highlights the importance of curvature, transitions, and parameter
coupling, while also exposing the challenges of real-world data and
engineering constraints.

As such, it serves as a primary application domain for
alignment-based information modelling.
\end{summary}
