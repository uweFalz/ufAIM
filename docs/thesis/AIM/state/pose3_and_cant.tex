% ============================================================
\section{Motivation}
% ============================================================

\begin{remark}
Planar railway alignment models describe intrinsic geometry using the
pose2 state introduced in \cref{def:pose2}.
\end{remark}

\begin{remark}
Operational railway systems are spatial systems involving:
\begin{itemize}
\item profile geometry,
\item cant,
\item local frame orientation,
\item vehicle-space interpretation,
\item and dynamically relevant quantities.
\end{itemize}
\end{remark}

\begin{remark}
This motivates an extended spatial railway state referred to as
\emph{pose3}.
\end{remark}

\begin{principle}[Runtime pose3 principle]
\label{princ:runtime-pose3}
Within AIM, pose3 is interpreted primarily as a runtime-evaluated
spatial railway state rather than as a persistent geometric storage
object.
\end{principle}

\begin{principle}[pose3 storage separation]
\label{princ:pose3-storage-separation}
pose3 is not persisted as independent geometry.

The persistent model stores only the quantities from which pose3 can be
reconstructed:
\begin{itemize}
\item intrinsic pose2 geometry,
\item profile information,
\item cant information,
\item gauge and convention data,
\item and metric/runtime context.
\end{itemize}
\end{principle}

\cref{eq:pose2-system}.

Within AIM, pose2 forms the persistent intrinsic reconstruction kernel,
whereas pose3 extends this state by runtime-evaluated spatial and
operational interpretation.
\end{remark}

\begin{remark}
pose3 depends on:
\begin{itemize}
\item pose2 reconstruction,
\item profile evaluation,
\item cant evaluation,
\item frame construction,
\item runtime metric services,
\item and runtime coordinate services.
\end{itemize}
\end{remark}

\begin{remark}
Conceptually:
\begin{itemize}
\item pose2 answers:
``where is the intrinsic alignment and how does it turn?''
\item profile answers:
``how does the alignment rise?''
\item cant answers:
``how is the rail plane rotated?''
\item pose3 answers:
``what operational spatial railway state results at runtime?''
\end{itemize}
\end{remark}

\section{Cant Representation}
% ============================================================

\subsection{Cant as Persistent Engineering Quantity}

\begin{definition}[Cant height difference]
\label{def:cant-height-difference}
Within AIM, persistent cant is represented as rail height difference:
\[
u(s).
\]
\end{definition}

\begin{remark}
This follows standard railway engineering practice, where cant is
typically specified as cross-level difference between rails.
\end{remark}

% ------------------------------------------------------------

\subsection{Derived Roll Angle}

\begin{definition}[Derived roll angle]
\label{def:derived-roll-angle}
Given a gauge value \(g_{\mathrm{gauge}}\), the corresponding roll angle
is:
\begin{equation}
\phi(s)
=
\arctan\!\left(
\frac{u(s)}{g_{\mathrm{gauge}}}
\right).
\label{eq:derived-roll-angle}
\end{equation}
\end{definition}

\begin{remark}
The roll angle is a derived runtime quantity rather than a persistent
primary variable.
\end{remark}

\begin{remark}
Within AIM, cant is interpreted geometrically as frame rotation,
independent of its engineering realization as rail height difference.
\end{remark}

% ------------------------------------------------------------

\subsection{Cant Evolution}

\begin{remark}
Cant evolution is governed by the canonical relation
\cref{eq:cant-evolution}.
\end{remark}

\begin{remark}
Cant-rate influences:
\begin{itemize}
\item dynamic comfort,
\item load transfer,
\item vehicle response,
\item and operational smoothness.
\end{itemize}
\end{remark}

% ============================================================
\section{Cant Reference Conventions}
% ============================================================

\begin{definition}[Cant reference convention]
\label{def:cant-reference-convention}
A cant convention specifies:
\begin{itemize}
\item the reference axis of rotation,
\item the sign convention of cant,
\item the relation between rail height difference and roll angle,
\item and the induced centerline displacement.
\end{itemize}
\end{definition}

\begin{remark}
Railway source data may define cant relative to:
\begin{itemize}
\item the centerline,
\item one rail,
\item a construction axis,
\item or another engineering reference.
\end{itemize}
\end{remark}

\begin{remark}
Within AIM, the intrinsic alignment centerline acts as canonical
reference geometry.
\end{remark}

\begin{remark}
Rail-based cant definitions must therefore be transformed consistently
during import, normalization, or runtime evaluation.
\end{remark}

\section{Pose3 Frames}
% ============================================================

\begin{definition}[Uncanted geometric frame]
\label{def:uncanted-frame}
The uncanted geometric frame is denoted by:
\[
F_0(s).
\]
\end{definition}

\begin{remark}
\(F_0(s)\) is constructed from the reconstructed spatial alignment
without cant rotation.
\end{remark}

% ------------------------------------------------------------

\begin{definition}[Railway frame]
\label{def:railway-frame}
The railway frame is denoted by:
\[
F_{\mathrm{rail}}(s).
\]
\end{definition}

\begin{remark}
\(F_{\mathrm{rail}}(s)\) is obtained by cant rotation of the geometric
frame \(F_0(s)\).
\end{remark}

% ------------------------------------------------------------

\begin{definition}[Vehicle frame]
\label{def:vehicle-frame}
A vehicle-dependent runtime frame may additionally be defined as:
\[
F_{\mathrm{veh}}(s).
\]
\end{definition}

\begin{remark}
Vehicle frames may depend on:
\begin{itemize}
\item suspension behaviour,
\item center-of-gravity location,
\item dynamic response,
\item and operational conditions.
\end{itemize}
\end{remark}

\section{Runtime pose3 State}
% ============================================================

\begin{definition}[Runtime pose3]
\label{def:runtime-pose3}
The runtime spatial railway state is the station-dependent evaluated
state:
\begin{equation}
\mathrm{pose3}(s)
=
(
\Gamma(s),
F_{\mathrm{rail}}(s),
\phi(s)
).
\label{eq:runtime-pose3}
\end{equation}
\end{definition}

\begin{remark}
Here:
\begin{itemize}
\item \(\Gamma(s)\) denotes the reconstructed spatial reference
trajectory,
\item \(F_{\mathrm{rail}}(s)\) denotes the cant-aware railway frame,
\item \(\phi(s)\) denotes the derived cant rotation.
\end{itemize}
\end{remark}

\begin{remark}
The railway frame is evaluated by the frame operator
\(\mathcal F\) introduced in \cref{def:frame-operator}.
\end{remark}

\begin{remark}
Within projected engineering geometry, the spatial trajectory
\(\Gamma(s)\) is assembled dynamically from:
\begin{itemize}
\item planar pose2 reconstruction,
\item profile evaluation,
\item metric interpretation,
\item and runtime coordinate services.
\end{itemize}
\end{remark}

\section{Persistent Model versus Runtime State}
% ============================================================

\begin{remark}
The persistent sparse model stores:
\begin{itemize}
\item intrinsic curvature geometry,
\item cant bands,
\item profile bands,
\item and sparse reconstruction parameters.
\end{itemize}
\end{remark}

\begin{remark}
Spatial pose3 states are reconstructed dynamically by runtime services.
\end{remark}

\begin{remark}
This avoids:
\begin{itemize}
\item redundant spatial storage,
\item synchronization instability,
\item and unnecessary duplication of derived geometry.
\end{itemize}
\end{remark}

\begin{remark}
The runtime interpretation may vary depending on:
\begin{itemize}
\item metric realization,
\item CRS interpretation,
\item level of detail,
\item vehicle-space interpretation,
\item and runtime evaluation strategy.
\end{itemize}
\end{remark}

The pose3 framework allows reinterpretation of railway alignment as
design of a spatial operational state trajectory.
\end{remark}

\begin{remark}
Schwerpunkt-Trassierung does not optimize the rail centerline alone.

Instead, it optimizes vehicle-relevant operational trajectories induced
by:
\begin{itemize}
\item curvature evolution,
\item cant evolution,
\item profile evolution,
\item frame interpretation,
\item and vehicle dynamics.
\end{itemize}
\end{remark}

\begin{remark}
In this interpretation, the railway alignment acts as a generator of
runtime vehicle-space states.
