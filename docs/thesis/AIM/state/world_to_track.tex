\begin{remark}
Railway alignments are formulated intrinsically using arc-length-based
coordinates.
\end{remark}

\begin{remark}
Real-world data is typically represented in external coordinate
reference systems such as:
\begin{itemize}
\item WGS84,
\item UTM,
\item Gauss--Krüger,
\item DBRef,
\item or local engineering CRS definitions.
\end{itemize}
\end{remark}

\begin{remark}
A transformation between intrinsic alignment coordinates and external
world coordinates is therefore required.
\end{remark}

\begin{remark}
Within AIM, this transformation is interpreted as a runtime evaluation
service connecting:
\begin{itemize}
\item intrinsic alignment states,
\item measured data,
\item visualization,
\item realization,
\item topology,
\item and operational geometry.
\end{itemize}
\end{remark}

\begin{principle}[Intrinsic-space primacy]
\label{princ:intrinsic-space-primacy}
Within AIM, intrinsic railway geometry is primary.

World-space coordinates are interpreted as external embeddings generated
through runtime evaluation.
\end{principle}

% ============================================================
\section{Track Coordinate System}
% ============================================================

\begin{definition}[Track space]
\label{def:track-space}
The track space is the intrinsic railway coordinate domain:
\[
\Omega_{\mathrm{track}}
=
\{(s,d,h)\}.
\]
It represents positions relative to an alignment rather than positions
in an external coordinate reference system.
\end{definition}

\begin{definition}[Track coordinate system]
\label{def:track-coordinate-system}
The intrinsic track coordinate system consists of:
\[
(s,d,h),
\]
where:
\begin{itemize}
\item \(s\) denotes intrinsic arc-length,
\item \(d\) denotes lateral offset,
\item \(h\) denotes vertical offset.
\end{itemize}
\end{definition}

\begin{remark}
Track coordinates belong to the intrinsic track-space domain
\(\Omega_{\mathrm{track}}\), introduced in
\cref{def:track-space}.
\end{remark}

\begin{remark}
In planar situations, only:
\[
(s,d)
\]
is required.
\end{remark}

\begin{remark}
Track coordinates are intrinsic alignment coordinates and must not be
confused with projected world coordinates.
\end{remark}

% ------------------------------------------------------------

\subsection{Intrinsic Geometry versus External Coordinates}

\begin{remark}
Intrinsic geometry is independent of any external coordinate reference
system.
\end{remark}

\begin{remark}
The same intrinsic alignment may be embedded into:
\begin{itemize}
\item projected engineering planes,
\item ellipsoidal world models,
\item local realization frames,
\item or visualization coordinate systems.
\end{itemize}
\end{remark}

\begin{remark}
Thus, intrinsic alignment geometry and external CRS representation form
distinct conceptual layers.
\end{remark}

% ------------------------------------------------------------

\subsection{Arc-Length and Stationing}

\begin{remark}
Intrinsic arc-length is not identical to engineering stationing.
\end{remark}

\begin{remark}
Operational kilometre systems may contain:
\begin{itemize}
\item station equations,
\item kilometre jumps,
\item overlapping station ranges,
\item topology-dependent references,
\item and operational branch relations.
\end{itemize}
\end{remark}

\begin{remark}
Stationing systems are operational reference structures layered on top
of intrinsic geometry.
\end{remark}

\begin{remark}
Consequently, the mapping:
\[
s
\leftrightarrow
\mathrm{km}
\]
is generally neither linear nor globally unique.
\end{remark}

\section{Track-to-World Transformation}
% ============================================================

\begin{definition}[Track-to-world mapping]
\label{def:state-track-to-world-runtime}
The forward transformation is:
\begin{equation}
x(s,d)
=
\gamma(s)+d\,\mathbf n(s).
\label{eq:state-track-to-world-runtime}
\end{equation}
\end{definition}

\begin{remark}
This relation specializes the canonical track-to-world operator
\(\mathcal P_{\mathrm{tw}}\) introduced in
\cref{def:track-to-world-operator}.
\end{remark}

% ------------------------------------------------------------

\subsection{Planar Tangent and Normal Directions}

\begin{remark}
In the planar case:
\[
\mathbf t(s)
=
(\cos\theta(s),\sin\theta(s)),
\]
and:
\[
\mathbf n(s)
=
(-\sin\theta(s),\cos\theta(s)).
\]
\end{remark}

% ------------------------------------------------------------

\subsection{Spatial Runtime Evaluation}

\begin{remark}
In spatial runtime evaluation, the mapping depends additionally on:
\begin{itemize}
\item profile,
\item cant,
\item frame construction,
\item metric interpretation,
\item and runtime realization context.
\end{itemize}
\end{remark}

\begin{remark}
Consequently, the world embedding:
\[
x(s,d,h)
\]
is generally generated dynamically from runtime pose3 evaluation.
\end{remark}

\subsection{Reference-Line Interpretation}

\begin{remark}
The lateral coordinate \(d\) depends on the chosen reference
interpretation.
\end{remark}

\begin{remark}
Possible reference structures include:

\item track center,

\section{World-to-Track Transformation}
% ============================================================

\begin{definition}[World-to-track projection]
\label{def:staete-world-to-track-runtime}
The inverse transformation is:
\begin{equation}
(s^\ast,d^\ast)
=
\arg\min_{s,d}
\|
x-(\gamma(s)+d\,\mathbf n(s))
\|.
\label{eq:state-world-to-track-runtime}
\end{equation}
\end{definition}

\begin{remark}
This relation specializes the canonical world-to-track operator
\(\mathcal P_{\mathrm{wt}}\) introduced in
\cref{def:world-to-track-operator}.
\end{remark}

\begin{remark}
The term projection is used operationally and does not necessarily imply
a unique orthogonal projection in the classical geometric sense.
\end{remark}

\begin{remark}
Within AIM, world-to-track projection is interpreted as an inverse
runtime evaluation problem.
\end{remark}

% ============================================================
\section{Coordinate Transformation versus Projection}
% ============================================================

\begin{remark}
Coordinate transformation and world-to-track projection are distinct
operations within AIM.
\end{remark}

\begin{remark}
Coordinate transformations map between world-space representations:
\[
\mathcal C
:
\Omega_{\mathrm{world}}
\rightarrow
\Omega_{\mathrm{world}},
\]
as introduced in
\cref{def:coordinate-transform-operator}.
\end{remark}

\begin{remark}
World-to-track projection maps between:
\[
\Omega_{\mathrm{world}}
\quad\text{and}\quad
\Omega_{\mathrm{track}}.
\]
\end{remark}

\begin{principle}[Projection distinction principle]
\label{princ:projection-distinction}
Coordinate transformation must not be confused with intrinsic
world-to-track projection.
\end{principle}

\begin{remark}
CRS transformations operate entirely within external coordinate
representations, whereas projection accesses intrinsic railway geometry.
\end{remark}

% ============================================================
\section{Numerical Evaluation}
% ============================================================

\begin{remark}
The inverse transformation is generally nonlinear.
\end{remark}

\begin{remark}
Typical numerical approaches include:
\begin{itemize}
\item nearest-point search,
\item local Newton iteration,
\item segment-wise projection,
\item topology-aware lookup,
\item and hybrid runtime strategies.
\end{itemize}
\end{remark}

\begin{remark}
Projection quality depends strongly on:
\begin{itemize}
\item curvature behaviour,
\item topology,
\item initialization,
\item runtime context,
\item and metric interpretation.
\end{itemize}
\end{remark}

% ============================================================
\section{Ambiguity and Context Dependence}
% ============================================================

\begin{remark}
World-to-track mapping is generally not globally unique.
\end{remark}

\begin{remark}
Ambiguities may occur in:
\begin{itemize}
\item loops,
\item switches and crossings,
\item parallel tracks,
\item closely spaced alignments,
\item station areas,
\item and topology-rich railway networks.
\end{itemize}
\end{remark}

\begin{remark}
Multiple intrinsic states may correspond to nearly identical world-space
locations.
\end{remark}

\begin{remark}
Consequently, projection is fundamentally context dependent.
\end{remark}

\begin{remark}
Runtime projection may require:
\begin{itemize}
\item track identity,
\item route identity,
\item topology context,
\item expected station range,
\item operational direction,
\item runtime interaction history,
\item vehicle context,
\item or network-state constraints.
\end{itemize}
\end{remark}

\begin{principle}[Projection ambiguity principle]
\label{princ:projection-ambiguity}
World-to-track projection cannot generally be interpreted as a purely
local geometric operation.
\end{principle}

% ============================================================
\section{Topology and Operational References}
% ============================================================

\begin{remark}
Intrinsic geometry alone is insufficient for operational railway
referencing.
\end{remark}

\begin{remark}
Operational interpretation additionally depends on:
\begin{itemize}
\item topology,
\item route membership,
\item kilometre references,
\item operational direction,
\item and network semantics.
\end{itemize}
\end{remark}

\begin{remark}
Thus, a complete projection context may require:
\[
(
x,
\mathcal T,
\mathcal R,
\mathcal O
),
\]
where:
\begin{itemize}
\item \(x\) denotes world-space input,
\item \(\mathcal T\) denotes topology context,
\item \(\mathcal R\) denotes route/reference context,
\item \(\mathcal O\) denotes operational context.
\end{itemize}
\end{remark}

% ============================================================
\section{Relation to Curvature}
% ============================================================

\begin{remark}
Projection stability depends directly on curvature behaviour.
\end{remark}

\begin{remark}
High curvature increases projection sensitivity, whereas low curvature
typically yields stable projection behaviour.
\end{remark}

\begin{remark}
Errors in intrinsic coordinates propagate nonlinearly into world space.
\end{remark}

\begin{remark}
Projection behaviour near transition regions may depend strongly on
higher-order curvature behaviour.
\end{remark}

% ============================================================
\section{Runtime Dependency}
% ============================================================

\begin{remark}
Within AIM, world--track transformation depends on runtime evaluation of
the underlying alignment representation.
\end{remark}

\begin{remark}
In particular:
\begin{itemize}
\item cGeom defines intrinsic geometry,
\item pose2 defines tangent evolution,
\item pose3 defines runtime spatial interpretation,
\item metric services define realized embedding,
\item CRS services define external representation,
\item and runtime operators evaluate resulting states.
\end{itemize}
\end{remark}

\begin{remark}
Projection therefore depends on the currently active runtime evaluation
pipeline.
\end{remark}

\begin{remark}
Different runtime configurations may yield different projection results,
for example:
\begin{itemize}
\item projected versus ellipsoidal evaluation,
\item target versus realized geometry,
\item low-resolution versus high-resolution runtime,
\item or simplified versus vehicle-aware interpretation.
\end{itemize}
\end{remark}

\begin{principle}[Runtime projection principle]
\label{princ:runtime-projection}
World--track transformation is a runtime evaluation problem operating on
alignment states rather than a static coordinate utility.
\end{principle}

% ============================================================
\section{LOD and Hybrid Evaluation}
% ============================================================

\begin{remark}
Exact world-to-track projection is computationally expensive.
\end{remark}

\begin{remark}
Large-scale systems therefore require:
\begin{itemize}
\item LOD-dependent evaluation,
\item approximate projection,
\item topology-aware filtering,
\item and hybrid runtime architectures.
\end{itemize}
\end{remark}

\begin{proposition}[Selective projection principle]
\label{prop:selective-projection}
Exact projection should only be evaluated where high precision is
required.
\end{proposition}

\begin{remark}
A hybrid strategy combines:
\begin{itemize}
\item precomputed samples for coarse lookup,
\item topology-aware candidate filtering,
\item cached runtime states,
\item and local parametric refinement for precision.
\end{itemize}
\end{remark}

\begin{remark}
Hybrid projection architectures are particularly important for:
\begin{itemize}
\item large railway networks,
\item interactive visualization,
\item realtime vehicle-space evaluation,
\item and multi-resolution runtime systems.
\end{itemize}
\end{remark}

\section{Connection to CRS and Realization}
% ============================================================

\begin{remark}
The complete external transformation chain is:
\[
\text{CRS}
\rightarrow
\Omega_{\mathrm{world}}
\rightarrow
\Omega_{\mathrm{track}}.
\]
\end{remark}

\begin{remark}
For high-precision applications, the alignment itself may be defined on
a reference ellipsoid rather than in a projected plane.
\end{remark}

\begin{remark}
Projection may also depend on realized geometry:
\[
\gamma_{\mathrm{real}}(s)
\neq
\gamma_{\mathrm{target}}(s).
\]
\end{remark}

\begin{remark}
Physical measurements should therefore reference realized rather than
purely analytical geometry.
\end{remark}

\begin{remark}
The distinction between:
\[
\text{intrinsic geometry}
\quad\text{and}\quad
\text{external embedding}
\]
is fundamental within AIM.
\end{remark}

\section{Key Insight}
% ============================================================

\begin{proposition}
World--track transformation is an inverse runtime service rather than a
static coordinate conversion.
\end{proposition}

\begin{proposition}
Within AIM, intrinsic alignment space is primary, whereas world
coordinates are interpreted as external embeddings of alignment states.
\end{proposition}

\begin{proposition}
Projection is generally:
\begin{itemize}
\item nonlinear,
\item ambiguous,
\item topology-dependent,
\item runtime-dependent,
\item and context-sensitive.
\end{itemize}
\end{proposition}

\section{Connection to AIM}
% ============================================================

\begin{summary}
World--track transformation connects intrinsic railway geometry with
external coordinate systems, topology, runtime evaluation, and measured
reality.

Its nonlinear, ambiguous, topology-dependent, and scale-dependent nature
makes it a central runtime component of the AIM framework.

Within AIM, projection is therefore interpreted not as static coordinate
conversion, but as contextual runtime reconstruction of intrinsic
railway states from external observations.

Efficient handling consequently requires:
\begin{itemize}
\item hybrid runtime architectures,
\item topology-aware projection strategies,
\item LOD-dependent evaluation,
\item sparse-state reconstruction,
\item and metric-aware runtime services.
\end{itemize}
\end{summary}
