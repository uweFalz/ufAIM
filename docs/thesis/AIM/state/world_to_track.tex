\section{World--Track Coordinate Transformation}

% ============================================================
\subsection{Motivation}
% ============================================================

\begin{remark}
Railway alignments are modelled in an intrinsic coordinate system
defined by arc-length.
\end{remark}

\begin{remark}
Real-world data, however, is given in global or engineering coordinate
systems such as WGS84, UTM, Gauss--Krüger, or local project reference
systems.
\end{remark}

\begin{remark}
Thus, a transformation between world coordinates and track coordinates
is required.
\end{remark}

\begin{remark}
Within AIM, world-to-track transformation is not merely a geometric
utility. It is a central runtime service connecting measured data,
visualization, validation, and alignment modelling.
\end{remark}

% ============================================================
\subsection{Track Coordinate System}
% ============================================================

\begin{definition}[Track coordinate system]
Let \(\gamma(s)\) denote the alignment curve parameterized by arc-length.

The track coordinate system at position \(s\) is defined by:
\begin{itemize}
\item longitudinal coordinate: \(s\),
\item lateral offset: \(d\),
\item vertical offset: \(h\).
\end{itemize}
\end{definition}

\begin{remark}
In the planar case, only \((s,d)\) is considered.
\end{remark}

\begin{remark}
Track coordinates are intrinsic to the alignment. They should not be
confused with projected world coordinates.
\end{remark}

\begin{remark}
The track coordinate system is not merely a coordinate convention.
It is the intrinsic reference structure of the alignment itself.
\end{remark}

\begin{remark}
Intrinsic track coordinates are not identical to engineering stationing.
Station equations, kilometre jumps, and operational reference systems
may introduce discontinuities unrelated to geometric arc-length.
\end{remark}

% ============================================================
\subsection{Forward Transformation (Track to World)}
% ============================================================

\begin{definition}[Track-to-world mapping]
Given \((s,d)\), the corresponding world position is:
\[
x(s,d)
=
\gamma(s)
+
d \cdot \mathbf{n}(s),
\]
where \(\mathbf{n}(s)\) is the normal vector of the alignment.
\end{definition}

\begin{remark}
The tangent vector is:
\[
\mathbf{t}(s)
=
(\cos\theta(s), \sin\theta(s)),
\]
and the normal vector:
\[
\mathbf{n}(s)
=
(-\sin\theta(s), \cos\theta(s)).
\]
\end{remark}

\begin{remark}
The lateral offset \(d\) depends on the chosen reference convention.
Possible references include:
\begin{itemize}
\item alignment centerline,
\item track center,
\item rail axis,
\item or engineering-specific offset definitions.
\end{itemize}
\end{remark}

\OPEN{
Clarify reference-line conventions for lateral offsets,
especially in multi-track and cant-dependent situations.
}

% ============================================================
\subsection{Inverse Transformation (World to Track)}
% ============================================================

\begin{definition}[World-to-track mapping]
Given a world point \(x\), the track coordinates \((s,d)\) are defined by:
\[
(s^\ast, d^\ast)
=
\arg\min_{s,d}
\| x - (\gamma(s) + d\,\mathbf{n}(s)) \|.
\]
\end{definition}

\begin{remark}
The term projection is used operationally here and does not necessarily
imply a unique orthogonal projection in the classical geometric sense.
\end{remark}

\centering

\begin{remark}
This is a nonlinear inverse problem on the alignment.
\end{remark}

\begin{remark}
In practice, it is solved by:
\begin{itemize}
\item nearest point search,
\item local Newton iteration,
\item segment-wise projection,
\item or hybrid lookup strategies.
\end{itemize}
\end{remark}

% ============================================================
\subsection{Ambiguity and Failure Modes}
% ============================================================

\begin{remark}
The world-to-track mapping is generally not globally unique and may be
ambiguous.
\end{remark}

\begin{remark}
A single world position may correspond to multiple valid intrinsic track
states.
\end{remark}

\begin{remark}
Ambiguities may occur in:
\begin{itemize}
\item loops,
\item switches and crossings,
\item parallel tracks,
\item closely spaced alignments,
\item self-near geometries,
\item or station areas.
\end{itemize}
\end{remark}

\begin{remark}
Therefore, world-to-track transformation must often be evaluated in a
context-aware way, using:
\begin{itemize}
\item track identity,
\item expected station range,
\item topology,
\item operational context,
\item or runtime interaction history.
\end{itemize}
\end{remark}

% ============================================================
\subsection{Local Approximation}
% ============================================================

\begin{remark}
For small offsets, the projection can be approximated by:
\[
d \approx (x - \gamma(s)) \cdot \mathbf{n}(s).
\]
\end{remark}

\begin{remark}
The arc-length coordinate is approximated by projection onto the tangent:
\[
s \approx \arg\min_s \| x - \gamma(s) \|.
\]
\end{remark}

% ============================================================
\subsection{Relation to Curvature}
% ============================================================

\begin{remark}
The transformation depends on curvature:
\begin{itemize}
\item high curvature increases projection sensitivity,
\item low curvature yields stable projections.
\end{itemize}
\end{remark}

\begin{remark}
Errors in \(s\) propagate nonlinearly into world space.
\end{remark}

\begin{remark}
World-to-track transformation therefore depends directly on the quality
and stability of the underlying alignment representation.
\end{remark}

% ============================================================
\subsection{Dependency on Runtime Evaluation}
% ============================================================

\begin{remark}
The world-to-track transformation depends on runtime evaluation of the
underlying sparse alignment model.
\end{remark}

\begin{remark}
Within AIM:
\begin{itemize}
\item cGeom defines the intrinsic alignment curve,
\item pose2 provides tangent and local orientation,
\item pose3 extends the local spatial state,
\item and sparse runtime operators evaluate the resulting geometry.
\end{itemize}
\end{remark}

\begin{remark}
Thus, world-to-track transformation is not an isolated geometric
operation, but part of the runtime evaluation architecture of the AIM
framework.
\end{remark}

% ============================================================
\subsection{Computational Cost}
% ============================================================

\begin{remark}
The inverse mapping \(x \mapsto (s,d)\) is computationally expensive:
\begin{itemize}
\item nonlinear,
\item iterative,
\item dependent on curve representation,
\item and potentially ambiguous.
\end{itemize}
\end{remark}

\begin{remark}
Thus, it must be used selectively in large-scale systems.
\end{remark}

\begin{proposition}[Selective projection principle]
Exact world-to-track projection should be performed only where its
precision is required.
\end{proposition}

% ============================================================
\subsection{Level-of-Detail Strategy}
% ============================================================

\begin{definition}[LOD-dependent projection]
The world-to-track transformation is evaluated depending on scale:
\begin{itemize}
\item coarse scale: approximate projection,
\item fine scale: exact projection.
\end{itemize}
\end{definition}

\begin{remark}
This reflects the fact that high precision is only required locally.
\end{remark}

\begin{remark}
LOD-dependent projection is not merely a visualization strategy, but a
computational necessity for scalable alignment-based systems.
\end{remark}

% ============================================================
\subsection{Hybrid Strategy}
% ============================================================

\begin{remark}
A hybrid approach combines:
\begin{itemize}
\item precomputed samples for fast lookup,
\item local parametric refinement for accuracy.
\end{itemize}
\end{remark}

\begin{remark}
This mirrors the hybrid modelling approach used for curvature.
\end{remark}

\begin{remark}
In this sense, world-to-track mapping is itself a hybrid runtime service:
coarse search identifies candidates, while local refinement provides
precision.
\end{remark}

% ============================================================
\subsection{Connection to CRS}
% ============================================================

\begin{remark}
The world coordinates depend on the chosen coordinate reference system
(CRS).
\end{remark}

\begin{remark}
Thus, the full transformation chain is:
\[
\text{CRS}
\;\rightarrow\;
\text{world}
\;\rightarrow\;
\text{track}.
\]
\end{remark}

\begin{remark}
For high-precision applications, the alignment curve \(\gamma(s)\) may
be defined directly on a reference ellipsoid.
This affects the definition of curvature and projection operators.
\end{remark}

% ============================================================
\subsection{Connection to Realization}
% ============================================================

\begin{remark}
The mapping depends on the realized alignment:
\[
\gamma_{\mathrm{real}}(s)
\neq
\gamma_{\mathrm{target}}(s).
\]
\end{remark}

\begin{remark}
Thus, world-to-track mapping should always refer to the realized
geometry whenever physical measurement data is involved.
\end{remark}

% ============================================================
\subsection{Key Insight}
% ============================================================

\begin{proposition}
The world-to-track transformation is the central interface between
geometric modelling and real-world data.
\end{proposition}

\begin{proposition}
World-to-track transformation is an inverse runtime service, not a
static coordinate conversion.
\end{proposition}

\begin{proposition}[Alignment-centric coordinate principle]
Within AIM, the intrinsic alignment space is primary.
World coordinates are interpreted as external embeddings of alignment
states rather than as the fundamental geometric domain.
\end{proposition}

% ============================================================
\subsection{Connection to AIM}
% ============================================================

\begin{summary}
The world--track transformation links the intrinsic alignment model
to external coordinate systems and measured data.

Its nonlinear, ambiguous, and scale-dependent nature makes it a critical
component for visualization, optimization, and data integration.

Within AIM, world-to-track transformation is interpreted as a runtime
evaluation service operating on sparse alignment structures rather than
as a purely geometric coordinate utility.

Efficient handling therefore requires hybrid and LOD-based strategies,
consistent with the overall AIM framework.
\end{summary}
