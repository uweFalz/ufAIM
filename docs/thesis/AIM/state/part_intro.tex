% \chapter{Railway State Model}

\begin{remark}
This part introduces the railway state model underlying the AIM
framework.
Instead of treating the track merely as a geometric curve, the state
model captures position, orientation, curvature, and cant in a unified
representation.
\end{remark}

\begin{remark}
The purpose of this part is to define the variables and transformations
required to connect intrinsic alignment modelling with dynamic railway
interpretation.
\end{remark}

\begin{remark}
This establishes the transition from purely geometric description
towards a state-based view of railway alignment.
\end{remark}
