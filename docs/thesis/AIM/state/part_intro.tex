% ============================================================
\section*{What do we already know?}
% ============================================================

\begin{remark}
The previous part established the intrinsic geometry of railway
alignments.
\end{remark}

\begin{remark}
Within AIM, geometry was shown to emerge from curvature evolution along
arc-length, rather than from stored coordinates or predefined geometric
primitives.
\end{remark}

\begin{remark}
Curvature space therefore provides a complete intrinsic description of
railway geometry.
\end{remark}

\begin{remark}
However, geometry alone does not yet constitute a railway state.
\end{remark}

% ============================================================
\section*{What is still missing?}
% ============================================================

\begin{remark}
A geometric description specifies what the alignment is.
\end{remark}

\begin{remark}
It does not yet specify how the alignment is represented,
reconstructed, interpreted, or evaluated during operation.
\end{remark}

\begin{remark}
Engineering applications require more than geometry alone.
\end{remark}

\begin{remark}
Position, direction, profile, cant, spatial orientation,
admissible motions, and runtime interpretation all require additional
state variables.
\end{remark}

\begin{remark}
A bridge therefore remains between intrinsic geometry and operational
railway behaviour.
\end{remark}

\begin{remark}
The missing element is a coherent state description.
\end{remark}

% ============================================================
\section*{What comes next?}
% ============================================================

\begin{remark}
This part introduces the railway state model underlying AIM.
\end{remark}

\begin{remark}
The first state layer is \(\mathrm{pose2}\), the intrinsic
reconstruction state generated from curvature.
\end{remark}

\begin{remark}
It provides the initial-value structure required to reconstruct
position and direction along the alignment.
\end{remark}

\begin{remark}
The second state layer is \(\mathrm{pose3}\), the spatial railway
state obtained through profile, cant, frame interpretation, and metric
realization.
\end{remark}

\begin{remark}
Importantly, \(\mathrm{pose3}\) describes the railway itself.
\end{remark}

\begin{remark}
Vehicle states are introduced only later as states evolving relative
to \(\mathrm{pose3}\).
\end{remark}

\begin{remark}
The chapters of this part establish the variables, frames, operators,
and transformations required to connect intrinsic geometry with
operational railway interpretation.
\end{remark}

\begin{principle}[Railway-State Principle]
\label{princ:railway-state}
Within AIM, railway alignment is interpreted through state evolution
rather than through coordinate geometry alone.
\end{principle}
