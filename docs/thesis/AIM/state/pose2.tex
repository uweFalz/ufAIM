\chapter{Pose2}

% ============================================================
\section{Motivation}
% ============================================================

\begin{remark}
In railway alignment modelling, the geometric state of the track at a
position along its length must be represented in a minimal, stable, and
intrinsic form.
\end{remark}

\begin{remark}
While full spatial railway modelling involves cant, profile,
realization, runtime services, and operational interpretation, many
fundamental geometric properties can already be expressed within a
planar intrinsic framework.
\end{remark}

\begin{remark}
The concept of \emph{pose2} provides this minimal intrinsic
representation.
\end{remark}

\begin{remark}
Within AIM, pose2 is not merely a geometric helper structure.
It forms the persistent intrinsic reconstruction kernel of the alignment
model.
\end{remark}

% ============================================================
\section{Definition of pose2}
% ============================================================

\begin{definition}[pose2]
A planar pose state at arc-length position \(s\) is defined as
\[
\mathrm{pose2}(s)
=
\bigl(
\gamma(s),
\theta(s)
\bigr).
\]
\label{def:pose2}
\end{definition}

\begin{remark}
The pose2 state describes:
\begin{itemize}
\item intrinsic planar position,
\item and intrinsic tangent orientation.
\end{itemize}
\end{remark}

\begin{remark}
The canonical pose2 system is defined in
\cref{eq:pose2-system}.
\end{remark}

% ============================================================
\section{Curvature-Driven Reconstruction}
% ============================================================

\begin{principle}[Curvature-driven reconstruction]
Within AIM, planar geometry is reconstructed intrinsically from
curvature evolution.
\end{principle}

\begin{remark}
The intrinsic reconstruction chain follows directly from
\cref{eq:curvature-relation,eq:state-evolution,eq:planar-tangent}.
\end{remark}

\begin{remark}
Thus, geometry is not treated as primary.
Instead, position and orientation emerge through integration of
curvature.
\end{remark}

% ============================================================
\section{Initial Conditions}
% ============================================================

\begin{definition}[Initial pose2 state]
A pose2 trajectory is uniquely determined by:
\[
\gamma(0),
\qquad
\theta(0),
\]
together with the curvature law
\[
\kappa(s).
\]
\label{def:initial-pose2}
\end{definition}

\begin{remark}
The curvature law acts as the geometric control function of planar state
evolution.
\end{remark}

% ============================================================
\section{Pose2 as Persistent Kernel}
% ============================================================

\begin{proposition}
Within AIM, pose2 forms the persistent intrinsic reconstruction kernel
of the alignment model.
\label{prop:pose2-persistent-kernel}
\end{proposition}

\begin{remark}
Pose2 is sufficient for persistent planar reconstruction of intrinsic
alignment geometry.
\end{remark}

\begin{remark}
Higher-level geometric and operational quantities are derived from
pose2 rather than stored redundantly.
\end{remark}

% ============================================================
\section{Connection to Sparse Alignment}
% ============================================================

\begin{remark}
Within the sparse alignment model, cGeom elements are interpreted as
local curvature operators contributing to reconstruction of the global
pose2 trajectory.
\end{remark}

\begin{remark}
Fixed and transition elements therefore do not primarily store geometry
itself, but rules governing geometric evolution.
\end{remark}

\begin{remark}
The sparse alignment model can thus be interpreted as a compact
integration program for reconstructing pose2 trajectories.
\end{remark}

% ============================================================
\section{Intrinsic Interpretation}
% ============================================================

\begin{remark}
pose2 separates:
\begin{itemize}
\item intrinsic geometric behaviour,
\item from derived geometric quantities.
\end{itemize}
\end{remark}

\begin{remark}
Curvature acts as intrinsic control quantity, whereas position and
orientation emerge through reconstruction.
\end{remark}

\begin{remark}
This separation forms one of the central structural principles of the
AIM framework.
\end{remark}

% ============================================================
\section{Relation to pose3}
% ============================================================

\begin{remark}
Within AIM, pose2 forms the persistent intrinsic reconstruction kernel,
whereas pose3 typically represents an extended runtime-evaluated spatial
state.
\end{remark}

\begin{remark}
pose3 is derived dynamically from:
\begin{itemize}
\item pose2 reconstruction,
\item cant evaluation,
\item profile evaluation,
\item frame construction,
\item and runtime coordinate services.
\end{itemize}
\end{remark}

\begin{remark}
This avoids redundancy and synchronization instability between planar,
cant, profile, and runtime representations.
\end{remark}

% ============================================================
\section{Canonical Interpretation}
% ============================================================

\begin{proposition}
Within AIM, intrinsic railway geometry is fundamentally curvature-driven.
\end{proposition}

\begin{proposition}
pose2 represents the canonical persistent intrinsic geometry state from
which higher-level runtime states are derived.
\end{proposition}

% ============================================================
\section{Connection to AIM}
% ============================================================

\begin{summary}
pose2 is the fundamental intrinsic planar state representation of
railway alignment.

Within AIM, pose2 forms the persistent reconstruction kernel from which
planar geometry is generated through curvature-driven integration.

Higher-dimensional runtime states such as pose3 are derived dynamically
from this kernel together with cant, profile, frame, and runtime
evaluation services.
\end{summary}
