\chapter{Pose2}

% ============================================================
\section{Motivation}
% ============================================================

\begin{remark}
In railway alignment modelling, the geometric state of the track at a
position along its length must be described in a minimal and consistent
way.
\end{remark}

\begin{remark}
While full spatial representation involves three-dimensional geometry,
cant, and dynamic interpretation, many fundamental properties can
already be captured in a planar setting.
\end{remark}

\begin{remark}
The concept of \emph{Pose2} provides such a minimal representation in
two dimensions.
\end{remark}

\begin{remark}
Within AIM, Pose2 is not merely a geometric helper structure.
It forms the persistent reconstruction kernel of the alignment model.
\end{remark}

% ============================================================
\section{Definition of Pose2}
% ============================================================

\begin{definition}[Pose2]
A planar pose at arc-length position $s$ is defined as the tuple
\[
\mathrm{Pose2}(s)
=
\bigl(
\gamma(s),
\theta(s)
\bigr),
\]
where:
\begin{itemize}
\item $\gamma(s)=(x(s),y(s))$ is the planar position,
\item $\theta(s)$ is the tangent direction.
\end{itemize}
\end{definition}

\begin{remark}
Pose2 describes the local geometric state of the alignment in the plane.
\end{remark}

% ============================================================
\section{Curvature-Driven Reconstruction}
% ============================================================

\begin{principle}[Curvature-driven reconstruction]
Within AIM, planar geometry is reconstructed from curvature by
successive integration:
\[
\kappa(s)
\rightarrow
\theta(s)
\rightarrow
\gamma(s).
\]
\end{principle}

% ============================================================
\section{Kinematic Relations}
% ============================================================

\begin{remark}
The evolution of Pose2 is governed by curvature.
\end{remark}

\begin{proposition}
The tangent direction satisfies:
\[
\frac{\dd\theta}{\dd s}
=
\kappa(s).
\]
\end{proposition}

\begin{proposition}
The planar position satisfies:
\[
\frac{\dd\gamma}{\dd s}
=
\begin{pmatrix}
\cos\theta(s) \\
\sin\theta(s)
\end{pmatrix}.
\]
\end{proposition}

\begin{remark}
Thus, geometry is not treated as primary.
Instead, it emerges from integration of curvature.
\end{remark}

% ============================================================
\section{Initial Conditions}
% ============================================================

\begin{definition}[Initial pose]
A Pose2 trajectory is uniquely determined by:
\[
\gamma(0),
\qquad
\theta(0),
\]
together with the curvature law
\[
\kappa(s).
\]
\end{definition}

\begin{remark}
The curvature law acts as the geometric control function of the planar
state evolution.
\end{remark}

% ============================================================
\section{Pose2 as Persistent Kernel}
% ============================================================

\begin{proposition}
Within AIM, Pose2 forms the persistent geometric reconstruction kernel
of the alignment model.
\end{proposition}

\begin{remark}
Pose2 is sufficient for persistent planar reconstruction of the
alignment geometry.
\end{remark}

\begin{remark}
Higher-level geometric and dynamic quantities are derived from Pose2
rather than stored redundantly.
\end{remark}

% ============================================================
\section{Connection to Sparse Alignment}
% ============================================================

\begin{remark}
Within the sparse alignment model, cGeom elements are interpreted as
local curvature operators contributing to the reconstruction of the
global Pose2 trajectory.
\end{remark}

\begin{remark}
Fixed and transition elements therefore do not primarily store geometry
itself, but rules governing geometric evolution.
\end{remark}

\begin{remark}
The sparse alignment model can thus be interpreted as a compact
integration program for reconstructing Pose2 trajectories.
\end{remark}

% ============================================================
\section{Interpretation}
% ============================================================

\begin{remark}
Pose2 separates:
\begin{itemize}
\item intrinsic geometric behaviour (curvature),
\item from derived geometric quantities (position and direction).
\end{itemize}
\end{remark}

\begin{remark}
This separation is one of the central structural ideas of the AIM
framework.
\end{remark}

% ============================================================
\section{Relation to Pose3}
% ============================================================

\begin{remark}
Within AIM, Pose3 is not treated as a persistent extension of Pose2,
but as a runtime-derived spatial evaluation state.
\end{remark}

\begin{remark}
Pose3 is derived dynamically from:
\begin{itemize}
\item Pose2 reconstruction,
\item cant evaluation,
\item profile evaluation,
\item and runtime coordinate services.
\end{itemize}
\end{remark}

\begin{remark}
This avoids redundancy and synchronization instability between planar,
cant, and profile representations.
\end{remark}

% ============================================================
\section{Summary}
% ============================================================

\begin{summary}
Pose2 is the fundamental planar state representation of railway
alignment.

Within AIM, Pose2 forms the persistent reconstruction kernel from which
planar geometry is generated through curvature-driven integration.

Higher-dimensional runtime states such as Pose3 are derived from this
kernel together with cant and profile information.
\end{summary}
