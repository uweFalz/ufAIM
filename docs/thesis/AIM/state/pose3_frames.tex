\chapter{Pose3 Frames and Spatial Reference Systems}

% ============================================================
\section*{Motivation}
% ============================================================

\begin{remark}
The pose3 state describes railway geometry together with orientation,
cant, and spatial interpretation.
\end{remark}

\begin{remark}
The state variables alone do not fully determine how the alignment is
interpreted in space. For this, local reference frames are required.
\end{remark}

\begin{remark}
Within AIM, frames form the operational bridge between:
\begin{itemize}
\item intrinsic geometry,
\item cant,
\item dynamics,
\item realization,
\item world--track projection,
\item and vehicle-space interpretation.
\end{itemize}
\end{remark}

% ============================================================
\section{Pose3 and Frames}
% ============================================================

\begin{remark}
The canonical pose3 state is defined in
\cref{eq:canonical-state}.
\end{remark}

\begin{remark}
A pose3 frame provides the local spatial interpretation of this state.
It is derived from the alignment and evaluated along arc-length.
\end{remark}

% ============================================================
\section{Tangent Direction}
% ============================================================

\begin{definition}[Tangent vector]
The tangent vector is:
\[
\mathbf t(s)=\gamma'(s).
\]
\end{definition}

\begin{remark}
For arc-length parameterization,
\[
\|\mathbf t(s)\|=1.
\]
\end{remark}

\begin{remark}
The tangent direction defines the local forward direction of motion.
\end{remark}

% ============================================================
\section{Lateral and Binormal Directions}
% ============================================================

\begin{definition}[Lateral direction]
The lateral direction is denoted by:
\[
\mathbf n(s),
\]
with
\[
\mathbf n(s)\cdot \mathbf t(s)=0.
\]
\end{definition}

\begin{definition}[Binormal direction]
The binormal direction is:
\[
\mathbf b(s)=\mathbf t(s)\times\mathbf n(s).
\]
\end{definition}

\begin{remark}
The lateral and binormal directions define the local cross-section
orientation of the railway frame.
\end{remark}

% ============================================================
\section{Pose3 Frame}
% ============================================================

\begin{definition}[Pose3 frame]
The local pose3 frame is:
\[
F(s)=
(\mathbf t(s),\mathbf n(s),\mathbf b(s)).
\]
\end{definition}

\begin{remark}
The pose3 frame defines the local operational coordinate system of the
railway alignment.
\end{remark}

\begin{remark}
It is a runtime-evaluated structure rather than an independently stored
geometric object.
\end{remark}

% ============================================================
\section{Cant Rotation}
% ============================================================

\begin{definition}[Cant rotation]
Cant is interpreted as a rotation around the tangent direction:
\[
R_\phi(s).
\]
\end{definition}

\begin{definition}[Cant-rotated frame]
The cant-rotated frame is:
\[
F_\phi(s)
=
(
\mathbf t(s),
\mathbf n_\phi(s),
\mathbf b_\phi(s)
).
\]
\end{definition}

\begin{remark}
The cant-rotated frame determines:
\begin{itemize}
\item rail elevations,
\item cross-level interpretation,
\item vehicle orientation,
\item effective gravity direction,
\item and platform geometry.
\end{itemize}
\end{remark}

% ============================================================
\section{Rail Plane}
% ============================================================

\begin{remark}
Gauge, rail elevation, and cross-level geometry are defined relative to
the cant-rotated frame.
\end{remark}

\begin{remark}
The rail plane is therefore not globally horizontal, but attached to the
local pose3 frame and its cant rotation.
\end{remark}

% ============================================================
\section{Frenet and Railway Frames}
% ============================================================

\begin{remark}
Classical Frenet frames are not fully adequate for railway alignment
modelling.
\end{remark}

\begin{remark}
In particular, Frenet frames become unstable in regions where
\[
\kappa(s)\to 0.
\]
\end{remark}

\begin{remark}
Railway modelling therefore requires frame formulations that remain
stable in straight and low-curvature sections.
\end{remark}

\begin{remark}
In practice, railway frames are often transported continuously along the
alignment rather than recomputed purely from differential geometry.
\end{remark}

% ============================================================
\section{Surface and Ellipsoid Frames}
% ============================================================

\begin{remark}
When alignments are defined on the Earth surface, local frames must
respect ellipsoid geometry.
\end{remark}

\begin{definition}[Surface-attached frame]
A surface-attached frame consists of:
\[
(\mathbf t,\mathbf n,\mathbf n_{\mathrm E}),
\]
where \(\mathbf n_{\mathrm E}\) denotes the ellipsoid surface normal.
\end{definition}

\begin{remark}
In this setting:
\begin{itemize}
\item curvature splits into geodesic and normal curvature,
\item gravity direction depends on Earth geometry,
\item and world-to-track projection becomes surface-dependent.
\end{itemize}
\end{remark}

% ============================================================
\section{Vehicle-Space Interpretation}
% ============================================================

\begin{remark}
The physically relevant object is not always the track centerline alone,
but the motion of the railway vehicle within space.
\end{remark}

\begin{remark}
Pose3 frames provide the spatial reference basis for:
\begin{itemize}
\item center-of-gravity motion,
\item vehicle envelopes,
\item bogie rotation,
\item wheelset orientation,
\item and platform clearance behaviour.
\end{itemize}
\end{remark}

\begin{remark}
Thus, vehicle-space interpretation is derived from alignment states and
their associated frames.
\end{remark}

% ============================================================
\section{Connection to World--Track Transformation}
% ============================================================

\begin{remark}
World--track projection relies directly on the local frame.
\end{remark}

\begin{remark}
In particular:
\begin{itemize}
\item tangent directions define longitudinal coordinates,
\item lateral directions define offsets,
\item and cant modifies the operational spatial interpretation.
\end{itemize}
\end{remark}

\begin{remark}
This connects pose3 frames to the canonical world--track relation in
\cref{eq:world-to-track}.
\end{remark}

% ============================================================
\section{Frame Continuity}
% ============================================================

\begin{remark}
Frame continuity is essential for:
\begin{itemize}
\item smooth visualization,
\item stable dynamics,
\item cant interpretation,
\item vehicle-space evaluation,
\item and numerical robustness.
\end{itemize}
\end{remark}

\begin{remark}
Discontinuous frame flips may lead to incorrect cant orientation,
unstable vehicle interpretation, and invalid projection behaviour.
\end{remark}

% ============================================================
\section{Key Insight}
% ============================================================

\begin{proposition}
Pose3 frames are operational runtime structures rather than purely
geometric differential objects.
\end{proposition}

\begin{remark}
They connect intrinsic railway geometry with realization, dynamics,
vehicle behaviour, and spatial interpretation.
\end{remark}
